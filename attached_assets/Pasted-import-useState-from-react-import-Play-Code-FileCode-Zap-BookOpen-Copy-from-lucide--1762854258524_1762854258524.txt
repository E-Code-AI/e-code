import { useState } from "react";
import { Play, Code, FileCode, Zap, BookOpen, Copy } from "lucide-react";
import { Button } from "./ui/button";
import { Badge } from "./ui/badge";
import { ScrollArea } from "./ui/scroll-area";
import { Tabs, TabsContent, TabsList, TabsTrigger } from "./ui/tabs";
import Editor from "@monaco-editor/react";
import { toast } from "sonner@2.0.3";

interface Template {
  id: string;
  name: string;
  language: string;
  code: string;
  description: string;
}

export function PlaygroundTab() {
  const [activeCode, setActiveCode] = useState(`// Welcome to E-Code Playground
// Write and test code snippets instantly

function fibonacci(n) {
  if (n <= 1) return n;
  return fibonacci(n - 1) + fibonacci(n - 2);
}

console.log("Fibonacci(10):", fibonacci(10));

// Try changing the code and click Run!
`);

  const [output, setOutput] = useState("");
  const [language, setLanguage] = useState("javascript");

  const templates: Template[] = [
    {
      id: "1",
      name: "Hello World",
      language: "javascript",
      code: `console.log("Hello, World!");`,
      description: "Simple hello world example",
    },
    {
      id: "2",
      name: "Fetch API",
      language: "javascript",
      code: `async function fetchData() {
  try {
    const response = await fetch('https://api.example.com/data');
    const data = await response.json();
    console.log(data);
  } catch (error) {
    console.error('Error:', error);
  }
}

fetchData();`,
      description: "Example API request",
    },
    {
      id: "3",
      name: "React Component",
      language: "typescript",
      code: `import React from 'react';

interface Props {
  name: string;
}

const Greeting: React.FC<Props> = ({ name }) => {
  return <h1>Hello, {name}!</h1>;
};

export default Greeting;`,
      description: "Simple React component",
    },
  ];

  const runCode = () => {
    setOutput("Running code...\n");
    setTimeout(() => {
      setOutput(
        `Running code...\n✓ Compilation successful\n\n--- Output ---\nFibonacci(10): 55\n\nExecution time: 12ms`
      );
    }, 500);
    toast.success("Code executed successfully");
  };

  const loadTemplate = (template: Template) => {
    setActiveCode(template.code);
    setLanguage(template.language);
    setOutput("");
    toast.info(`Loaded: ${template.name}`);
  };

  return (
    <div className="h-full flex flex-col bg-background">
      {/* Header */}
      <div className="border-b p-4">
        <div className="flex items-center justify-between">
          <div>
            <h2 className="flex items-center gap-2">
              <Code className="w-5 h-5" />
              Playground
            </h2>
            <p className="text-sm text-muted-foreground mt-1">
              Test code snippets and experiment with ideas
            </p>
          </div>
          <div className="flex gap-2">
            <Button variant="outline" size="sm" onClick={() => toast.success("Share link copied to clipboard", { description: "Share your playground with others" })}>
              <Copy className="w-4 h-4 mr-2" />
              Share
            </Button>
            <Button onClick={runCode}>
              <Play className="w-4 h-4 mr-2" />
              Run
            </Button>
          </div>
        </div>
      </div>

      <div className="flex-1 flex overflow-hidden">
        {/* Editor Area */}
        <div className="flex-1 flex flex-col">
          <Tabs defaultValue="code" className="flex-1 flex flex-col">
            <div className="border-b px-3">
              <TabsList className="h-10">
                <TabsTrigger value="code">Code</TabsTrigger>
                <TabsTrigger value="output">Output</TabsTrigger>
                <TabsTrigger value="templates">Templates</TabsTrigger>
              </TabsList>
            </div>

            <TabsContent value="code" className="flex-1 m-0 p-0">
              <Editor
                height="100%"
                language={language}
                value={activeCode}
                onChange={(value) => setActiveCode(value || "")}
                theme="vs-dark"
                options={{
                  minimap: { enabled: false },
                  fontSize: 14,
                  lineNumbers: "on",
                  wordWrap: "on",
                  automaticLayout: true,
                  scrollBeyondLastLine: false,
                  padding: { top: 16, bottom: 16 },
                }}
              />
            </TabsContent>

            <TabsContent value="output" className="flex-1 m-0 p-0">
              <div className="h-full bg-[#1e1e1e] text-gray-300 p-4 font-mono text-sm overflow-auto">
                {output || "Run your code to see output here..."}
              </div>
            </TabsContent>

            <TabsContent value="templates" className="flex-1 m-0 p-0">
              <ScrollArea className="h-full">
                <div className="p-4 grid gap-3">
                  {templates.map((template) => (
                    <div
                      key={template.id}
                      className="p-4 border rounded-lg hover:bg-accent cursor-pointer"
                      onClick={() => loadTemplate(template)}
                    >
                      <div className="flex items-start justify-between mb-2">
                        <div className="flex items-center gap-2">
                          <FileCode className="w-4 h-4 text-blue-500" />
                          <h4 className="text-sm">{template.name}</h4>
                        </div>
                        <Badge variant="secondary" className="text-xs">
                          {template.language}
                        </Badge>
                      </div>
                      <p className="text-xs text-muted-foreground mb-3">
                        {template.description}
                      </p>
                      <code className="text-xs bg-muted px-2 py-1 rounded block overflow-x-auto">
                        {template.code.split("\n")[0]}...
                      </code>
                    </div>
                  ))}
                </div>
              </ScrollArea>
            </TabsContent>
          </Tabs>
        </div>

        {/* Sidebar - Quick Actions */}
        <div className="w-64 border-l p-4 space-y-4">
          <div>
            <h3 className="text-sm mb-3">Language</h3>
            <div className="grid grid-cols-2 gap-2">
              {["JavaScript", "TypeScript", "Python", "Go", "Rust", "Java"].map(
                (lang) => (
                  <Button
                    key={lang}
                    variant={
                      language === lang.toLowerCase() ? "default" : "outline"
                    }
                    size="sm"
                    onClick={() => setLanguage(lang.toLowerCase())}
                  >
                    {lang}
                  </Button>
                )
              )}
            </div>
          </div>

          <div className="p-3 bg-muted rounded-lg">
            <h3 className="flex items-center gap-2 text-sm mb-2">
              <Zap className="w-4 h-4" />
              Quick Tips
            </h3>
            <ul className="text-xs text-muted-foreground space-y-2">
              <li>• Press Ctrl+Enter to run</li>
              <li>• Use console.log() for output</li>
              <li>• Auto-save enabled</li>
              <li>• Share your code via URL</li>
            </ul>
          </div>

          <div className="p-3 bg-muted rounded-lg">
            <h3 className="flex items-center gap-2 text-sm mb-2">
              <BookOpen className="w-4 h-4" />
              Examples
            </h3>
            <div className="space-y-2">
              <Button variant="ghost" size="sm" className="w-full justify-start">
                Arrays & Loops
              </Button>
              <Button variant="ghost" size="sm" className="w-full justify-start">
                Async/Await
              </Button>
              <Button variant="ghost" size="sm" className="w-full justify-start">
                DOM Manipulation
              </Button>
              <Button variant="ghost" size="sm" className="w-full justify-start">
                API Requests
              </Button>
            </div>
          </div>
        </div>
      </div>
    </div>
  );
}
