5.3 DEEP LINKING INCOMPLET
Fichier: desktop/main.js (lignes 677-693)

Problème:

function handleDeepLink(url) {
  if (url.startsWith('ecode://')) {
    mainWindow.webContents.send('deep-link', url);  // URL brute!
  }
}

Solution:

function handleDeepLink(url) {
  const parsed = new URL(url);
  const data = {
    action: parsed.pathname.split('/')[1],  // 'project', 'file', etc.
    params: Object.fromEntries(parsed.searchParams),
    path: parsed.pathname,
  };
  mainWindow.webContents.send('deep-link', data);
}

5.4 GESTION MULTI-FENÊTRES MANQUANTE
Fichier: desktop/main.js (lignes 33-36)

Problème: Une seule variable mainWindow globale.

Solution:

class WindowManager {
  constructor() {
    this.windows = new Map();
  }
  
  createWindow(type, options) {
    const win = new BrowserWindow(options);
    const id = Date.now().toString();
    this.windows.set(id, { window: win, type });
    return id;
  }
  
  getWindow(id) {
    return this.windows.get(id)?.window;
  }
}

5.5 AUTO-UPDATE SANS RETRY
Fichier: desktop/main.js (lignes 836-854)

Problème: Pas de logique de retry pour les mises à jour échouées.

Solution:

let updateRetries = 0;
const MAX_UPDATE_RETRIES = 3;

autoUpdater.on('error', (error) => {
  if (updateRetries < MAX_UPDATE_RETRIES) {
    updateRetries++;
    setTimeout(() => autoUpdater.checkForUpdates(), 5000 * updateRetries);
  } else {
    dialog.showErrorBox('Update Error', 'Failed to download update. Please try again later.');
  }
});

5.6 CLEANUP À LA FERMETURE MANQUANT
Fichier: desktop/main.js (lignes 1250-1253)

Problème:

app.on('before-quit', () => {
  // Vide!
});

Solution:

app.on('before-quit', async () => {
  // Fermer les watchers de fichiers
  fileWatcher?.close();
  
  // Déconnecter WebSocket
  wsConnection?.close();
  
  // Nettoyer les fichiers temporaires
  await fs.rm(tempDir, { recursive: true, force: true });
  
  // Sauvegarder l'état
  store.set('windowState', mainWindow.getBounds());
});
