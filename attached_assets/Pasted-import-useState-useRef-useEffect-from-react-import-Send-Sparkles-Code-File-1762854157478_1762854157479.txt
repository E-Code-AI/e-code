// Utility function to calculate relative time
const getRelativeTime = (timestamp: Date): string => {
  const now = new Date();
  const diffInSeconds = Math.floor((now.getTime() - timestamp.getTime()) / 1000);
  
  if (diffInSeconds < 60) {
    return "Just now";
  }
  
  const diffInMinutes = Math.floor(diffInSeconds / 60);
  if (diffInMinutes < 60) {
    return diffInMinutes === 1 ? "1 minute ago" : `${diffInMinutes} minutes ago`;
  }
  
  const diffInHours = Math.floor(diffInMinutes / 60);
  if (diffInHours < 24) {
    return diffInHours === 1 ? "1 hour ago" : `${diffInHours} hours ago`;
  }
  
  const diffInDays = Math.floor(diffInHours / 24);
  if (diffInDays === 1) {
    return "Yesterday";
  }
  if (diffInDays < 7) {
    return `${diffInDays} days ago`;
  }
  
  const diffInWeeks = Math.floor(diffInDays / 7);
  if (diffInWeeks < 4) {
    return diffInWeeks === 1 ? "1 week ago" : `${diffInWeeks} weeks ago`;
  }
  
  const diffInMonths = Math.floor(diffInDays / 30);
  if (diffInMonths < 12) {
    return diffInMonths === 1 ? "1 month ago" : `${diffInMonths} months ago`;
  }
  
  const diffInYears = Math.floor(diffInDays / 365);
  return diffInYears === 1 ? "1 year ago" : `${diffInYears} years ago`;
};

      timestamp: new Date(Date.now() - 35 * 60 * 1000),
    },
  ];

  useEffect(() => {
    if (scrollRef.current) {
      scrollRef.current.scrollTop = scrollRef.current.scrollHeight;
    }
  }, [messages]);

  // Auto-refresh timestamps every 30 seconds
  const [, setTimestampRefresh] = useState(0);
  useEffect(() => {
    const interval = setInterval(() => {
      setTimestampRefresh((prev) => prev + 1);
    }, 30000); // Refresh every 30 seconds

    return () => clearInterval(interval);
  }, []);

  // Extract the send logic into a reusable function
  const sendMessage = async (messageText: string) => {
    if (!messageText.trim() || isLoading) return;

    const userMessage: MessageItem = {
      id: Date.now().toString(),
      type: "user",
      content: messageText,
      timestamp: new Date(),
      userMessage: true,
    };

    setMessages((prev) => [...prev, userMessage]);
    const userInput = messageText;
    setInput("");
    setIsLoading(true);

    // Start Sentry transaction for monitoring
    const transaction = startAgentTransaction('ai-agent-request', {
      projectId: projectId || 'unknown',
      aiModel: selectedModel,
    });
    const startTime = Date.now();

    // Call the AGENT backend API (creates files automatically like Replit)
    try {
      if (!session?.access_token) {
        toast.error("Please sign in to use the AI agent");
        setIsLoading(false);
        transaction.setStatus('unauthenticated');
        transaction.finish();
        return;
      }

      if (!projectId) {
        toast.error("No project selected");
        setIsLoading(false);
        transaction.setStatus('failed_precondition');
        transaction.finish();
        return;
      }

      // Show "Agent is thinking..." status
      const thinkingMessage: MessageItem = {
        id: (Date.now() + 1).toString(),
        type: "status",
        status: { text: "Agent is thinking...", active: true },
        timestamp: new Date(),
      };
      setMessages((prev) => [...prev, thinkingMessage]);

      const result = await safeMutation(`${SERVER_URL}/chat/agent`, {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
          Authorization: `Bearer ${session.access_token}`,
        },
        body: JSON.stringify({
          message: userInput,
          projectId: projectId,
          userId: session.user?.id,
          aiModel: selectedModel,
          autonomyLevel,
          appTesting,
          webSearch,
          imageGeneration,
        }),
      });

      if (!result.ok) {
        // Log error to Sentry with context
        captureAgentError(new Error(result.error || "AI Agent API error"), {
          projectId,
          aiModel: selectedModel,
          userInput,
        });
        throw new Error(result.error || "AI Agent unavailable");
      }

      const data = result.data;
      
      // Remove thinking status
      setMessages((prev) => prev.filter((m) => m.id !== thinkingMessage.id));

      // Process actions and add messages
      const actionsToDisplay: MessageItem[] = [];
      let filesCreatedCount = 0;
      let filesEditedCount = 0;
      const createdFiles: any[] = [];

      for (const action of data.actions || []) {
        if (action.type === "message") {
          actionsToDisplay.push({
            id: Date.now().toString() + Math.random(),
            type: "text",
            content: action.content,
            timestamp: new Date(),
          });
        } else if (action.type === "create_file") {
          // Show action in chat
          actionsToDisplay.push({
            id: Date.now().toString() + Math.random(),
            type: "action",
            action: {
              type: "created",
              description: `Created ${action.path}`,
              duration: "0.5s",
            },
            timestamp: new Date(),
          });

          // Actually create the file via callback
          if (onCodeGenerate) {
            const fileName = action.path.split('/').pop() || action.path;
            onCodeGenerate(action.content, fileName);
          }

          filesCreatedCount++;
          createdFiles.push(action);
        } else if (action.type === "edit_file") {
          // Show action in chat
          actionsToDisplay.push({
            id: Date.now().toString() + Math.random(),
            type: "action",
            action: {
              type: "edited",
              description: `Edited ${action.path}`,
              duration: "0.3s",
            },
            timestamp: new Date(),
          });

          // Actually edit the file via callback
          if (onFileModify) {
            onFileModify(action.path, action.content);
          }

          filesEditedCount++;
        } else if (action.type === "install_package") {
          actionsToDisplay.push({
            id: Date.now().toString() + Math.random(),
            type: "action",
            action: {
              type: "installed",
              description: `Installed ${action.package}`,
              duration: "1.2s",
            },
            timestamp: new Date(),
          });
        } else if (action.type === "start_server") {
          // Auto-start server
          if (onStartServer) {
            onStartServer();
          }
          actionsToDisplay.push({
            id: Date.now().toString() + Math.random(),
            type: "status",
            status: { text: "Starting development server...", active: true },
            timestamp: new Date(),
          });
        } else if (action.type === "run_command") {
          actionsToDisplay.push({
            id: Date.now().toString() + Math.random(),
            type: "action",
            action: {
              type: "planned",
              description: `Running: ${action.command}`,
              duration: "2s",
            },
            timestamp: new Date(),
          });
        }
      }

      // Add summary message
      if (data.summary) {
        actionsToDisplay.push({
          id: Date.now().toString() + Math.random(),
          type: "text",
          content: `✅ ${data.summary}`,
          timestamp: new Date(),
        });
      }

      // Show next steps
      if (data.nextSteps && data.nextSteps.length > 0) {
        actionsToDisplay.push({
          id: Date.now().toString() + Math.random(),
          type: "text",
          content: `**Next steps:**\n${data.nextSteps.map((step: string, i: number) => `${i + 1}. ${step}`).join('\n')}`,
          timestamp: new Date(),
        });
      }

      setMessages((prev) => [...prev, ...actionsToDisplay]);

      // Log success to Sentry
      const duration = Date.now() - startTime;
      logAgentInteraction({
        projectId: projectId || 'unknown',
        aiModel: selectedModel,
        action: 'chat_completion',
        success: true,
        filesCreated: filesCreatedCount,
        filesEdited: filesEditedCount,
        duration,
      });

      // Auto-open preview if files were created or edited
      if (filesCreatedCount > 0 || filesEditedCount > 0) {
        const totalChanges = filesCreatedCount + filesEditedCount;
        toast.success(`${filesCreatedCount > 0 ? `Created ${filesCreatedCount} file(s)` : ''} ${filesEditedCount > 0 ? `Edited ${filesEditedCount} file(s)` : ''}`.trim(), {
          description: "Opening preview...",
        });
        
        // Notify parent about files created
        if (onFilesCreated && createdFiles.length > 0) {
          onFilesCreated(createdFiles);
        }

        // Auto-open preview after a short delay
        setTimeout(() => {
          if (onOpenPreview) {
            onOpenPreview();
          }
        }, 1000);
      }

      transaction.setStatus('ok');
    } catch (error: any) {
      console.error("AI Agent error:", error);
      
      // Log to Sentry with full context
      captureAgentError(error, {
        projectId,
        aiModel: selectedModel,
        userInput,
        autonomyLevel,
      });
      
      toast.error(error.message || "Failed to get AI response");
      
      const errorMessage: MessageItem = {
        id: (Date.now() + 1).toString(),
        type: "text",
        content: "Sorry, I encountered an error processing your request. Please try again.",
        timestamp: new Date(),
      };
      setMessages((prev) => [...prev, errorMessage]);
      
      transaction.setStatus('unknown_error');
    } finally {
      setIsLoading(false);
      transaction.finish();
    }
  };

  // Process initialPrompt from Homepage/Dashboard on first load
  useEffect(() => {
    if (initialPrompt && !initialPromptProcessed.current && session?.access_token) {
      initialPromptProcessed.current = true;
      
      // Show loading toast
      toast.info("🚀 Starting your project with AI Agent...");
      
      // Automatically send the initial prompt
      setTimeout(() => {
        sendMessage(initialPrompt);
      }, 800);
    }
  }, [initialPrompt, session]);

  // Wrapper function for the send button
  const handleSend = async () => {
    await sendMessage(input);
  };

    setMessages((prev) =>
      prev.map((msg) =>
        msg.id === id ? { ...msg, expanded: !msg.expanded } : msg

    );
  };

  const toggleSection = (sectionId: string) => {

      ...prev,
      [sectionId]: !prev[sectionId],
    }));
  };

  const handleClearChat = () => {
    setMessages([
      {
        id: "1",
        type: "text",
        content: "👋 Hi! I'm E-Code Agent, your intelligent coding partner. What would you like to create today?",

    setShowDemoMessages(false);

    setShowDemoMessages(true);
    setMessages([...messages, ...demoMessages]);
  };

      case "searched":
        return <Search className="w-3.5 h-3.5" />;
      case "decided":
        return <Brain className="w-3.5 h-3.5" />;
      case "opened":
        return <FileText className="w-3.5 h-3.5" />;
      case "edited":
        return <FileEdit className="w-3.5 h-3.5" />;
      case "replaced":
        return <RefreshCw className="w-3.5 h-3.5" />;
      case "uninstalled":
        return <Package className="w-3.5 h-3.5" />;
      case "installed":
        return <Package className="w-3.5 h-3.5" />;
      case "planned":
        return <Target className="w-3.5 h-3.5" />;
      case "architected":
        return <Target className="w-3.5 h-3.5" />;
      case "transformed":
        return <Cog className="w-3.5 h-3.5" />;
      case "implementing":
        return <Cog className="w-3.5 h-3.5" />;
      case "created":
        return <Target className="w-3.5 h-3.5" />;

    if (msg.agentBadge) {

            <div className="flex items-center gap-2">
              <Undo2 className="w-4 h-4 text-muted-foreground" />
              <span className="text-sm">{msg.content}</span>
            </div>
            <div className="flex items-center gap-2">
              <Badge variant="secondary" className="bg-orange-100 text-orange-700 dark:bg-orange-900/30 dark:text-orange-400">
                <Sparkles className="w-3 h-3 mr-1" />
                {msg.agentBadge}
              </Badge>
              <Button variant="ghost" size="sm" className="h-7 w-7 p-0">
                <RefreshCw className="w-3.5 h-3.5" />
              </Button>
            </div>
          </div>
        </div>
      );
    }

    // User message
    if (msg.type === "user") {
      return (
        <div key={msg.id} className="flex justify-end mb-4">
          <div className="max-w-[80%]">
            <div className="bg-blue-500 text-white rounded-2xl rounded-tr-sm px-4 py-2.5">
              <p className="text-sm leading-relaxed">{msg.content}</p>
            </div>
            <div className="flex items-center justify-end gap-1.5 mt-1 px-2">
              <span className="text-[10px] text-muted-foreground">
                {getRelativeTime(msg.timestamp)}
              </span>
              <Circle className="w-1.5 h-1.5 text-muted-foreground fill-muted-foreground" />
            </div>
          </div>
        </div>

    }

    // Section header (collapsible)
    if (msg.type === "section-header" && msg.section) {
      const sectionId = msg.id;
      const isOpen = sectionsOpen[sectionId] ?? msg.section.defaultOpen ?? false;
      
      return (
        <div key={msg.id} className="mb-3">
          <button
            onClick={() => toggleSection(sectionId)}

          >
            <span className="text-sm font-medium">{msg.section.title}</span>

                <div className="flex items-center gap-2">
                  <Timer className="w-4 h-4 text-muted-foreground" />
                  <span className="text-sm">
                    {msg.expanded ? "Show less" : msg.workSummary.duration}
                  </span>
                </div>
                <ChevronDown className={`w-4 h-4 transition-transform ${msg.expanded ? 'rotate-180' : ''}`} />
              </CollapsibleTrigger>
              
              <CollapsibleContent>
                <div className="px-3 pb-3 space-y-2">
                  <div className="flex items-center justify-between py-2">
                    <span className="text-sm">Time worked</span>
                    <span className="text-sm font-medium">{msg.workSummary.timeWorked}</span>
                  </div>
                  <div className="flex items-center justify-between py-2">
                    <span className="text-sm">Work done</span>
                    <span className="text-sm font-medium">{msg.workSummary.workDone}</span>
                  </div>
                  <div className="flex items-center justify-between py-2">
                    <span className="text-sm">Items read</span>
                    <span className="text-sm font-medium">{msg.workSummary.itemsRead}</span>
                  </div>
                  <div className="flex items-center justify-between py-2">
                    <div className="flex items-center gap-2">
                      <span className="text-sm">Agent Usage</span>
                      <ChevronDown className="w-3 h-3 text-muted-foreground" />
                    </div>
                    <span className="text-sm font-medium">{msg.workSummary.agentUsage}</span>
                  </div>
                </div>
              </CollapsibleContent>
            </div>
          </Collapsible>
          <div className="flex items-center gap-1.5 mt-1.5 ml-2">
            <Circle className="w-1.5 h-1.5 text-muted-foreground fill-muted-foreground" />

              {getRelativeTime(msg.timestamp)}
            </span>
          </div>
        </div>
      );
    }

    // Work duration
    if (msg.type === "work-duration") {
      return (
        <div key={msg.id} className="mb-3">
          <button
            onClick={() => toggleSection(msg.id)}
            className="w-full flex items-center justify-between p-3 bg-muted/30 hover:bg-muted/50 rounded-lg transition-colors border"
          >
            <div className="flex items-center gap-2">
              <Timer className="w-4 h-4 text-muted-foreground" />
              <span className="text-sm">{msg.workDuration}</span>
            </div>
            <ChevronDown className={`w-4 h-4 transition-transform ${sectionsOpen[msg.id] ? 'rotate-180' : ''}`} />
          </button>
          <div className="flex items-center justify-end gap-1.5 mt-1.5 mr-2">
            <span className="text-[10px] text-muted-foreground">
              {getRelativeTime(msg.timestamp)}
            </span>
            <Circle className="w-1.5 h-1.5 text-muted-foreground fill-muted-foreground" />
          </div>
        </div>
      );
    }

    // Test result (Testing your app)
    if (msg.type === "test-result" && msg.testResult) {
      return (
        <div key={msg.id} className="mb-4">
          <Collapsible
            open={msg.expanded}
            onOpenChange={() => toggleExpandable(msg.id)}
          >
            <CollapsibleTrigger className="w-full flex items-center justify-between p-3 hover:bg-muted/30 rounded-lg transition-colors border-b">
              <div className="flex items-center gap-2">
                <ChevronDown className={`w-4 h-4 transition-transform ${msg.expanded ? '' : '-rotate-90'}`} />
                <span className="text-sm font-medium">{msg.testResult.title}</span>
              </div>
              <ChevronUp className="w-4 h-4 text-muted-foreground" />
            </CollapsibleTrigger>
            <CollapsibleContent>
              <div className="mt-3">
                {/* Test header */}
                <div className="flex items-center justify-between p-3 bg-muted/50 rounded-t-lg border border-b-0">
                  <div className="flex items-center gap-2">
                    <RefreshCw className="w-4 h-4 animate-spin text-muted-foreground" />
                    <span className="text-sm">{msg.testResult.testName}</span>
                  </div>
                  <div className="flex items-center gap-2">
                    <Button variant="ghost" size="sm" className="h-7 w-7 p-0">
                      <PlayCircle className="w-4 h-4" />
                    </Button>
                    <Button variant="ghost" size="sm" className="h-7 w-7 p-0">
                      <Eye className="w-4 h-4" />
                    </Button>
                  </div>
                </div>
                
                {/* Browser preview */}
                <div className="relative bg-muted/30 rounded-b-lg border border-t-0 overflow-hidden" style={{ minHeight: '400px' }}>
                  <div className="absolute inset-0 flex items-center justify-center bg-neutral-300 dark:bg-neutral-700">
                    <p className="text-sm text-neutral-600 dark:text-neutral-300">
                      {msg.testResult.errorMessage}
                    </p>
                  </div>
                  {/* Purple button in bottom right */}
                  <div className="absolute bottom-4 right-4">
                    <div className="w-10 h-10 rounded-lg bg-purple-500 flex items-center justify-center shadow-lg">
                      <Circle className="w-5 h-5 text-white fill-white" />
                    </div>
                  </div>
                </div>
                
                {/* Explanation text */}
                {msg.testResult.explanation && (
                  <div className="mt-3 p-4 bg-muted/30 rounded-lg border">
                    <p className="text-sm text-muted-foreground leading-relaxed">
                      {msg.testResult.explanation}

    if (msg.type === "test-action" && msg.testAction) {

        <div key={msg.id} className="mb-2">
          <div className="flex items-start gap-2 text-sm text-muted-foreground">
            <RefreshCw className="w-3.5 h-3.5" />
            <span>
              <span className="capitalize">{msg.testAction.actionType}</span>{" "}
              <code className="text-xs bg-muted px-1.5 py-0.5 rounded">{msg.testAction.description}</code>

    if (msg.type === "action" && msg.action) {

          <div className="flex items-start gap-2 text-sm text-muted-foreground">
            {renderActionIcon(msg.action.type)}
            <span>
              <span className="capitalize">{msg.action.type}</span>{" "}
              {msg.action.description}
              {msg.action.duration && <span className="text-muted-foreground/60"> ({msg.action.duration})</span>}

    }

    if (msg.type === "task-item" && msg.taskItem) {
      const isHovered = hoveredTaskItem === msg.id;
      
      return (
        <div 
          key={msg.id} 
          className="mb-2 group"
          onMouseEnter={() => setHoveredTaskItem(msg.id)}
          onMouseLeave={() => setHoveredTaskItem(null)}
        >
          <div className="flex items-start justify-between gap-2 text-sm">
            <div className="flex items-start gap-2 flex-1">

          <div className="p-4 bg-muted/50 rounded-lg border">
            <div className="flex items-center gap-2 mb-2">
              <CheckCircle2 className="w-4 h-4 text-green-600 dark:text-green-500" />
              <span className="text-xs text-muted-foreground">

                <RotateCcw className="w-3 h-3 mr-1.5" />
                Rollback here

              >

                Changes
              </Button>
              <Button 
                variant="outline" 
                size="sm" 
                className="h-8 text-xs"
                onClick={() => toast.success("Preview updated", { description: "Rendering latest changes" })}
              >
                <Eye className="w-3 h-3 mr-1.5" />
                Preview
              </Button>
            </div>
          </div>
          <div className="flex items-center justify-end gap-1.5 mt-1.5 mr-2">
            <span className="text-[10px] text-muted-foreground">
              {getRelativeTime(msg.timestamp)}
            </span>
            <Circle className="w-1.5 h-1.5 text-muted-foreground fill-muted-foreground" />
          </div>
        </div>
      );
    }

    // Task list
    if (msg.type === "task-list" && msg.taskList) {
      return (
        <div key={msg.id} className="mb-4">
          <Collapsible
            open={msg.expanded}
            onOpenChange={() => toggleExpandable(msg.id)}
            className="border rounded-lg bg-background"
          >
            <CollapsibleTrigger className="w-full flex items-center justify-between p-3 hover:bg-muted/50">
              <div className="flex items-center gap-2">
                <ChevronRight className={`w-4 h-4 transition-transform ${msg.expanded ? 'rotate-90' : ''}`} />
                <span className="text-sm font-medium">{msg.taskList.title}</span>
              </div>
              <div className="flex items-center gap-2">
                <span className="text-xs text-muted-foreground">{msg.taskList.progress}</span>
                <RefreshCw className="w-4 h-4 text-muted-foreground" />
              </div>
            </CollapsibleTrigger>
            <CollapsibleContent>
              <div className="px-3 pb-3 space-y-2.5">
                {msg.taskList.tasks.map((task, idx) => (
                  <div key={idx} className="flex items-center gap-2.5 text-sm">
                    <Checkbox 
                      checked={task.done} 
                      className="h-4 w-4 rounded-full"
                    />
                    <span className={task.done ? "line-through text-muted-foreground" : ""}>
                      {task.text}
                    </span>
                    {task.hasSettings && (
                      <Settings className="w-3.5 h-3.5 text-muted-foreground ml-auto" />
                    )}
                  </div>
                ))}
                {msg.taskList.showMore && msg.taskList.hiddenCount && (
                  <button 
                    className="flex items-center gap-2 text-sm text-muted-foreground hover:text-foreground pl-6 pt-1"
                    onClick={() => toast.info("Expanding tasks", { description: `Showing ${msg.taskList!.hiddenCount} more tasks` })}
                  >
                    <ArrowDown className="w-3 h-3" />
                    <span>Show {msg.taskList.hiddenCount} more</span>
                  </button>
                )}
              </div>
            </CollapsibleContent>
          </Collapsible>
        </div>
      );
    }

    // Status (Working...)
    if (msg.type === "status" && msg.status) {
      return (
        <div key={msg.id} className="mb-4">
          <div className="flex items-center justify-between p-3 bg-purple-50 dark:bg-purple-950/20 rounded-lg border border-purple-200 dark:border-purple-800">
            <div className="flex items-center gap-2">
              <div className="w-2 h-2 bg-purple-500 rounded-full animate-pulse" />
              <span className="text-sm text-purple-700 dark:text-purple-400">{msg.status.text}</span>
            </div>
            {msg.status.active && (
              <Button 
                variant="outline" 
                size="sm" 
                className="h-7 text-xs border-purple-300 dark:border-purple-700"
                onClick={() => toast.success("Process stopped", { description: "Agent work cancelled" })}
              >
                <Square className="w-3 h-3 mr-1.5" />
                Stop
              </Button>
            )}
          </div>
        </div>
      );
    }

    // Code block
    if (msg.type === "code" && msg.code) {
      return (
        <div key={msg.id} className="mb-4">
          <div className="bg-muted/50 rounded-lg border overflow-hidden">
            <div className="flex items-center justify-between px-4 py-2 bg-muted border-b">
              <span className="text-xs font-medium">{msg.code.language}</span>
              <Button 
                variant="ghost" 
                size="sm" 
                className="h-6 text-xs"
                onClick={() => {
                  navigator.clipboard.writeText(msg.code!.content);
                  toast.success("Code copied to clipboard");
                }}
              >
                Copy
              </Button>
            </div>
            <pre className="p-4 text-sm overflow-x-auto">
              <code className="text-orange-600 dark:text-orange-400">{msg.code.content}</code>
            </pre>
          </div>
        </div>
      );
    }

    // Regular text message
    if (msg.type === "text") {
      const isExpandable = msg.expandable;
      const content = msg.content || "";
      
      if (isExpandable && content === "") {
        return (
          <button 
            key={msg.id} 
            onClick={() => toggleExpandable(msg.id)}
            className="flex items-center gap-2 text-sm text-blue-600 dark:text-blue-400 hover:underline mb-3"
          >
            <ArrowDown className="w-3 h-3" />
            <span>Show 9 more</span>
          </button>
        );
      }
      
      if (isExpandable) {
        return (
          <div key={msg.id} className="mb-4">
            <Collapsible open={msg.expanded} onOpenChange={() => toggleExpandable(msg.id)}>
              <CollapsibleTrigger className="flex items-center gap-2 text-sm text-muted-foreground hover:text-foreground mb-2">
                <ChevronUp className={`w-3 h-3 transition-transform ${msg.expanded ? '' : 'rotate-180'}`} />
                <span>Show {msg.expanded ? 'less' : 'more'}</span>
              </CollapsibleTrigger>
              <CollapsibleContent>
                <div className="text-sm leading-relaxed whitespace-pre-wrap">
                  {content}
                </div>
              </CollapsibleContent>
            </Collapsible>
          </div>
        );
      }

      return (
        <div key={msg.id} className="mb-3">
          <div className="text-sm leading-relaxed whitespace-pre-wrap">
            {content}
          </div>
        </div>
      );
    }

    return null;
  };

  return (
    <div className="h-full flex flex-col bg-background border-l">
      {/* Header */}
      <div className="flex items-center justify-between px-4 py-3 border-b bg-gradient-to-r from-purple-50 to-blue-50 dark:from-purple-950/20 dark:to-blue-950/20">
        <div className="flex items-center gap-3">
          <div className="relative">
            <div className="w-8 h-8 rounded-lg bg-gradient-to-br from-purple-500 to-blue-600 flex items-center justify-center">
              <Brain className="w-4 h-4 text-white" />
            </div>
            <div className="absolute -bottom-1 -right-1 w-3 h-3 bg-green-500 border-2 border-background rounded-full"></div>
          </div>
          <div>
            <h3 className="font-medium">E-Code Agent</h3>
            <p className="text-xs text-muted-foreground">AI-Powered Development</p>
          </div>
        </div>
        <div className="flex gap-1">
          <Badge variant="outline" className="text-xs">
            {selectedModel === "gpt-4" && "GPT-4"}
            {selectedModel === "gpt-4-turbo" && "GPT-4 Turbo"}
            {selectedModel === "claude-3-5-sonnet" && "Claude 3.5"}
            {selectedModel === "claude-3-sonnet" && "Claude 3"}
          </Badge>
          <Badge variant={agentMode === "autonomous" ? "default" : "secondary"} className="text-xs">
            {agentMode === "autonomous" ? (
              <>
                <Zap className="w-3 h-3 mr-1" />
                Autonomous
              </>
            ) : (
              "Chat Mode"
            )}
          </Badge>
          <DropdownMenu>
            <DropdownMenuTrigger asChild>
              <Button variant="ghost" size="sm" className="h-8 w-8 p-0">
                <MoreVertical className="w-4 h-4" />
              </Button>
            </DropdownMenuTrigger>
            <DropdownMenuContent align="end">
              <DropdownMenuItem onClick={() => setAgentMode("chat")}>
                Chat Mode
              </DropdownMenuItem>
              <DropdownMenuItem onClick={() => setAgentMode("autonomous")}>
                <Zap className="w-4 h-4 mr-2" />
                Autonomous Mode
              </DropdownMenuItem>
              <DropdownMenuSeparator />
              <DropdownMenuItem onClick={loadDemoMessages}>
                <Eye className="w-4 h-4 mr-2" />
                Load Demo Messages
              </DropdownMenuItem>
              <DropdownMenuItem>
                <Settings className="w-4 h-4 mr-2" />
                Agent Settings
              </DropdownMenuItem>
              <DropdownMenuItem onClick={handleClearChat}>
                <Trash2 className="w-4 h-4 mr-2" />
                Clear Chat
              </DropdownMenuItem>
            </DropdownMenuContent>
          </DropdownMenu>
        </div>
      </div>

      {/* Pinned In Progress Tasks */}
      {messages.some(msg => msg.type === "task-list" && msg.taskList?.title === "In progress tasks") && (
        <div className="border-b bg-background">
          <div className="px-4 py-3">
            {messages
              .filter(msg => msg.type === "task-list" && msg.taskList?.title === "In progress tasks")
              .map((msg) => renderMessage(msg))}
          </div>
        </div>
      )}

      {/* Messages */}
      <div className="flex-1 overflow-hidden">
        <ScrollArea className="h-full">
          <div className="px-4 py-4" ref={scrollRef}>
            {/* Show previous events button */}
            {messages.length > 5 && (
              <button className="w-full flex items-center justify-center gap-2 p-3 mb-4 border rounded-lg hover:bg-muted/50 transition-colors">
                <ChevronUp className="w-4 h-4" />
                <span className="text-sm">Show previous events</span>
              </button>
            )}
            
            <div className="space-y-2">
              {messages
                .filter(msg => !(msg.type === "task-list" && msg.taskList?.title === "In progress tasks"))
                .map((msg) => renderMessage(msg))}
            </div>
          </div>
        </ScrollArea>
      </div>

      {/* Input */}
      <div className="p-3 border-t bg-background">
        <div className="relative border rounded-lg bg-background">
          {/* Top toolbar */}
          <div className="flex items-center justify-between px-3 py-2 border-b">
            <div className="flex items-center gap-1">
              <DropdownMenu>
                <DropdownMenuTrigger asChild>
                  <Button variant="ghost" size="sm" className="h-7 gap-1 px-2">
                    <PlayCircle className="w-3.5 h-3.5" />
                    <span className="text-xs">Build</span>
                    <ChevronDown className="w-3 h-3" />
                  </Button>
                </DropdownMenuTrigger>
                <DropdownMenuContent align="start">
                  <DropdownMenuItem>
                    <PlayCircle className="w-4 h-4 mr-2" />
                    Build
                  </DropdownMenuItem>
                  <DropdownMenuItem>
                    <Upload className="w-4 h-4 mr-2" />
                    Deploy
                  </DropdownMenuItem>
                  <DropdownMenuItem>
                    <Wrench className="w-4 h-4 mr-2" />
                    Test
                  </DropdownMenuItem>
                </DropdownMenuContent>
              </DropdownMenu>
              
              <Separator orientation="vertical" className="h-4" />
              
              <Button variant="ghost" size="sm" className="h-7 w-7 p-0">
                <Sparkles className="w-3.5 h-3.5" />
              </Button>
              
              <Popover open={showAttachmentMenu} onOpenChange={setShowAttachmentMenu}>
                <PopoverTrigger asChild>
                  <Button variant="ghost" size="sm" className="h-7 w-7 p-0">
                    <Paperclip className="w-3.5 h-3.5" />
                  </Button>
                </PopoverTrigger>
                <PopoverContent align="start" className="w-64 p-0">
                  <div className="p-1.5">
                    <div className="px-2 py-1.5 text-xs font-medium text-muted-foreground">
                      Add attachments
                    </div>
                    
                    <button
                      onClick={() => {
                        setShowAttachmentMenu(false);
                        const input = document.createElement('input');
                        input.type = 'file';
                        input.click();
                      }}
                      className="w-full flex items-center gap-3 px-3 py-2.5 rounded-md hover:bg-accent transition-colors text-left"
                    >
                      <Upload className="w-4 h-4" />
                      <span className="text-sm">Upload a file</span>
                    </button>

                    <button
                      onClick={() => {
                        setShowAttachmentMenu(false);
                        setShowFigmaDialog(true);
                      }}
                      className="w-full flex items-center gap-3 px-3 py-2.5 rounded-md hover:bg-accent transition-colors text-left"
                    >
                      <svg className="w-4 h-4" viewBox="0 0 24 24" fill="none">
                        <path d="M12 12L12 24M12 12L6 12C3.79086 12 2 10.2091 2 8C2 5.79086 3.79086 4 6 4L12 4L12 12ZM12 12L18 12C20.2091 12 22 10.2091 22 8C22 5.79086 20.2091 4 18 4L12 4L12 12ZM12 12L12 0M12 12C12 14.2091 10.2091 16 8 16C5.79086 16 4 14.2091 4 12C4 9.79086 5.79086 8 8 8L12 12Z" fill="url(#figma-gradient)" />
                        <defs>
                          <linearGradient id="figma-gradient" x1="2" y1="2" x2="22" y2="22" gradientUnits="userSpaceOnUse">
                            <stop stopColor="#F24E1E" />
                            <stop offset="0.25" stopColor="#FF7262" />
                            <stop offset="0.5" stopColor="#A259FF" />
                            <stop offset="0.75" stopColor="#1ABCFE" />
                            <stop offset="1" stopColor="#0ACF83" />
                          </linearGradient>
                        </defs>
                      </svg>
                      <span className="text-sm">Attach a Figma design</span>
                    </button>
                  </div>
                </PopoverContent>
              </Popover>
            </div>

            <div className="flex items-center gap-1">
              <Popover open={showAgentTools} onOpenChange={setShowAgentTools}>
                <PopoverTrigger asChild>
                  <Button variant="ghost" size="sm" className="h-7 w-7 p-0">
                    <Settings className="w-3.5 h-3.5" />
                  </Button>
                </PopoverTrigger>
                <PopoverContent className="w-[340px] p-4" align="end">
                  <div className="space-y-4">
                    <div className="flex items-center gap-2">
                      <Wrench className="w-4 h-4" />
                      <h3 className="font-semibold">Agent Tools</h3>
                    </div>
                    
                    <div className="flex items-start justify-between gap-4">
                      <div className="flex-1 space-y-0.5">
                        <Label className="text-sm font-medium">App testing</Label>
                        <p className="text-xs text-muted-foreground leading-snug">
                          Automatically test your app
                        </p>
                      </div>
                      <Switch 
                        checked={appTesting} 
                        onCheckedChange={setAppTesting}
                      />
                    </div>

                    <Separator />

                    <div className="space-y-2.5">
                      <div className="space-y-0.5">
                        <Label className="text-sm font-medium">Autonomy level</Label>
                        <p className="text-xs text-muted-foreground leading-snug">
                          How independently should the agent work
                        </p>
                      </div>
                      <div className="grid grid-cols-4 gap-1.5">
                        {(["Low", "Medium", "High", "Max"] as const).map((level) => (
                          <Button
                            key={level}
                            variant={autonomyLevel === level ? "default" : "outline"}
                            size="sm"
                            className="text-xs h-8 px-2"
                            onClick={() => setAutonomyLevel(level)}
                          >
                            {level}
                          </Button>
                        ))}
                      </div>
                    </div>

                    <Separator />

                    <div className="space-y-2.5">
                      <div className="space-y-0.5">
                        <Label className="text-sm font-medium">AI Model</Label>
                        <p className="text-xs text-muted-foreground leading-snug">
                          Select the AI model to use
                        </p>
                      </div>
                      <div className="grid grid-cols-2 gap-1.5">
                        <Button
                          variant={selectedModel === "gpt-4" ? "default" : "outline"}
                          size="sm"
                          className="text-xs h-8 px-2"
                          onClick={() => setSelectedModel("gpt-4")}
                        >
                          GPT-4
                        </Button>
                        <Button
                          variant={selectedModel === "gpt-4-turbo" ? "default" : "outline"}
                          size="sm"
                          className="text-xs h-8 px-2"
                          onClick={() => setSelectedModel("gpt-4-turbo")}
                        >
                          GPT-4 Turbo
                        </Button>
                        <Button
                          variant={selectedModel === "claude-3-5-sonnet" ? "default" : "outline"}
                          size="sm"
                          className="text-xs h-8 px-2"
                          onClick={() => setSelectedModel("claude-3-5-sonnet")}
                        >
                          Claude 3.5
                        </Button>
                        <Button
                          variant={selectedModel === "claude-3-sonnet" ? "default" : "outline"}
                          size="sm"
                          className="text-xs h-8 px-2"
                          onClick={() => setSelectedModel("claude-3-sonnet")}
                        >
                          Claude 3
                        </Button>
                      </div>
                    </div>

                    <Separator />

                    <div className="flex items-start justify-between gap-4">
                      <div className="flex-1 space-y-0.5">
                        <Label className="text-sm font-medium flex items-center gap-1.5">
                          <Globe className="w-3.5 h-3.5" />
                          Web search
                        </Label>
                        <p className="text-xs text-muted-foreground leading-snug">
                          Search the web for information
                        </p>
                      </div>
                      <Switch 
                        checked={webSearch} 
                        onCheckedChange={setWebSearch}
                      />
                    </div>

                    <Separator />

                    <div className="flex items-start justify-between gap-4">
                      <div className="flex-1 space-y-0.5">
                        <Label className="text-sm font-medium flex items-center gap-1.5">
                          <ImageIcon className="w-3.5 h-3.5" />
                          Image generation
                        </Label>
                        <p className="text-xs text-muted-foreground leading-snug">
                          Generate images with AI
                        </p>
                      </div>
                      <Switch 
                        checked={imageGeneration} 
                        onCheckedChange={setImageGeneration}
                      />
                    </div>
                  </div>
                </PopoverContent>
              </Popover>
            </div>
          </div>

          {/* Textarea */}
          <div className="relative">
            <Textarea
              value={input}
              onChange={(e) => setInput(e.target.value)}
              onKeyDown={(e) => {
                if (e.key === "Enter" && !e.shiftKey) {
                  e.preventDefault();
                  handleSend();
                }
              }}
              placeholder="Make, test, iterate..."
              className="min-h-[80px] resize-none border-0 focus-visible:ring-0 focus-visible:ring-offset-0 rounded-none rounded-b-lg"
            />
          </div>

          {/* Send button */}
          <Button
            onClick={handleSend}
            disabled={!input.trim() || isLoading}
            size="icon"
            className="absolute bottom-2 right-2 h-8 w-8 rounded-md bg-gradient-to-r from-purple-500 to-blue-600 hover:from-purple-600 hover:to-blue-700"
          >
            <Send className="w-4 h-4" />
          </Button>
        </div>

        {/* Bottom hint */}
        <p className="text-xs text-muted-foreground mt-2 px-1">
          <Sparkles className="w-3 h-3 inline mr-1" />
          E-Code Agent can make mistakes. Always review generated code.
        </p>
      </div>

      {/* Figma Import Dialog */}
      <Dialog open={showFigmaDialog} onOpenChange={setShowFigmaDialog}>
        <DialogContent className="sm:max-w-[540px]">
          <DialogHeader>
            <div className="flex items-center gap-2">
              <svg className="w-5 h-5" viewBox="0 0 24 24" fill="none">
                <path d="M12 12L12 24M12 12L6 12C3.79086 12 2 10.2091 2 8C2 5.79086 3.79086 4 6 4L12 4L12 12ZM12 12L18 12C20.2091 12 22 10.2091 22 8C22 5.79086 20.2091 4 18 4L12 4L12 12ZM12 12L12 0M12 12C12 14.2091 10.2091 16 8 16C5.79086 16 4 14.2091 4 12C4 9.79086 5.79086 8 8 8L12 12Z" fill="url(#figma-gradient-dialog)" />
                <defs>
                  <linearGradient id="figma-gradient-dialog" x1="2" y1="2" x2="22" y2="22" gradientUnits="userSpaceOnUse">
                    <stop stopColor="#F24E1E" />
                    <stop offset="0.25" stopColor="#FF7262" />
                    <stop offset="0.5" stopColor="#A259FF" />
                    <stop offset="0.75" stopColor="#1ABCFE" />
                    <stop offset="1" stopColor="#0ACF83" />
                  </linearGradient>
                </defs>
              </svg>
              <DialogTitle>Import Figma design</DialogTitle>
              <Badge variant="secondary" className="ml-1">Beta</Badge>
            </div>
          </DialogHeader>

          <div className="space-y-4 py-4">
            <div className="space-y-2">
              <Label htmlFor="figma-url">Figma frame URL</Label>
              <Input
                id="figma-url"
                placeholder="https://www.figma.com/file/..."
                value={figmaUrl}
                onChange={(e) => setFigmaUrl(e.target.value)}
                className="h-10"
              />
            </div>

            <div className="flex gap-2 p-3 bg-muted/50 rounded-lg">
              <HelpCircle className="w-4 h-4 mt-0.5 flex-shrink-0 text-muted-foreground" />
              <div className="text-xs text-muted-foreground">
                <span className="font-medium">How to find your frame URL:</span> In Figma Design, right click on the frame you want to import, then click "Copy/Paste as" and finally "Copy link to selection".
              </div>
            </div>
          </div>

          <DialogFooter>
            <Button
              variant="outline"
              onClick={() => {
                setShowFigmaDialog(false);
                setFigmaUrl("");
              }}
            >
              Cancel
            </Button>
            <Button
              disabled={!figmaUrl.trim()}
              onClick={() => {
                console.log("Importing Figma design:", figmaUrl);
                setShowFigmaDialog(false);
                setFigmaUrl("");
              }}
            >
              Attach to prompt
            </Button>
          </DialogFooter>
        </DialogContent>
      </Dialog>
    </div>
  );
}
