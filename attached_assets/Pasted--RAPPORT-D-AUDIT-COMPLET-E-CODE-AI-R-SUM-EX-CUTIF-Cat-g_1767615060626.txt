Problème 12 : Race conditions sur l'auto-save
Fichier: /home/user/e-code/client/src/components/CodeEditor.tsx
Lignes: 109-144

Description: Plusieurs modifications rapides déclenchent des sauvegardes concurrentes sans queue.

Solution:

// Ajouter un système de queue:
import { useMemo } from 'react';
import { debounce } from 'lodash';

const saveQueue = useRef<string[]>([]);
const isSaving = useRef(false);

const processQueue = async () => {
  if (isSaving.current || saveQueue.current.length === 0) return;
  
  isSaving.current = true;
  const content = saveQueue.current.pop();
  saveQueue.current = []; // Clear pending saves
  
  try {
    await saveFile(content);
  } finally {
    isSaving.current = false;
    if (saveQueue.current.length > 0) {
      processQueue();
    }
  }
};

const debouncedSave = useMemo(
  () => debounce((content: string) => {
    saveQueue.current.push(content);
    processQueue();
  }, 2000),
  []
);

async function revokeToken(token: string, expiresAt: Date): Promise<void> {
  const ttl = Math.ceil((expiresAt.getTime() - Date.now()) / 1000);

export function useTerminalStream(sessionId: string) {
  const [output, setOutput] = useState<string>('');
  
  useEffect(() => {
    const ws = new WebSocket(`${WS_URL}/terminal/${sessionId}`);
    
    ws.onmessage = (event) => {
      const data = JSON.parse(event.data);
      if (data.type === 'output') {
        setOutput(prev => prev + data.content);
      }
    };
    
    return () => ws.close();
  }, [sessionId]);
  
  return output;
}

import { encode } from 'gpt-tokenizer';

const MAX_CONTEXT_TOKENS = 8000;

function truncateHistory(messages: Message[]): Message[] {
  let totalTokens = 0;
  const truncated: Message[] = [];
  
  // Parcourir du plus récent au plus ancien
  for (let i = messages.length - 1; i >= 0; i--) {
    const tokens = encode(messages[i].content).length;
    if (totalTokens + tokens > MAX_CONTEXT_TOKENS) break;
    totalTokens += tokens;
    truncated.unshift(messages[i]);
  }
  
  return truncated;
}
