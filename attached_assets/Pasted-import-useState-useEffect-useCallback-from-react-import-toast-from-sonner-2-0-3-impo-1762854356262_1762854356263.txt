                Edit as JSON
              </Button>
              <Button variant="outline" size="sm" onClick={handleEditAsEnv}>
                Edit as .env
              </Button>
            </div>
          </div>
        </TabsContent>

        <TabsContent value="account" className="flex-1 mt-0 overflow-auto">
          <div className="p-6 space-y-4">
            <Alert className="bg-orange-50 border-orange-200">
              <AlertDescription className="text-orange-900">
                You've reached the limit of 10 linked secrets. Please unlink some secrets before adding more.
              </AlertDescription>
            </Alert>

            <div className="bg-accent/30 border rounded-lg p-4">
              <div className="flex items-center justify-between">
                <p className="text-sm">Account secrets are only visible to you.</p>
                <Button variant="outline" size="sm">
                  Link to this App
                </Button>
              </div>
              <p className="text-sm text-muted-foreground mt-2">
                Linked secrets will stay in sync with your Account Secrets.
              </p>
            </div>

            <div className="bg-card border rounded-lg overflow-hidden">
              <div className="divide-y">
                {accountSecrets.map(secret => renderSecret(secret, false))}
              </div>
            </div>
          </div>
        </TabsContent>
      </Tabs>

      {/* New Secret Dialog */}
      <Dialog open={showNewSecretDialog} onOpenChange={setShowNewSecretDialog}>
        <DialogContent>
          <DialogHeader>
            <DialogTitle>Add New Secret</DialogTitle>
            <DialogDescription>
              Create a new environment variable for your application
            </DialogDescription>
          </DialogHeader>
          
          <div className="space-y-4">
            <div className="space-y-2">
              <Label>Key</Label>
              <Input
                placeholder="API_KEY"
                value={newSecretKey}
                onChange={(e) => setNewSecretKey(e.target.value)}
              />
            </div>
            <div className="space-y-2">
              <Label>Value</Label>
              <Input
                type="password"
                placeholder="your-secret-value"
                value={newSecretValue}
                onChange={(e) => setNewSecretValue(e.target.value)}
              />
            </div>
            <div className="space-y-2">
              <Label>Category (Optional)</Label>
              <Select value={newSecretCategory} onValueChange={setNewSecretCategory}>
                <SelectTrigger>
                  <SelectValue placeholder="Select category" />
                </SelectTrigger>
                <SelectContent>
                  {categories.filter(c => c !== "all").map(cat => (
                    <SelectItem key={cat} value={cat}>
                      {cat.charAt(0).toUpperCase() + cat.slice(1)}
                    </SelectItem>
                  ))}
                </SelectContent>
              </Select>
            </div>
            <div className="space-y-2">
              <Label>Tags (Optional)</Label>
              <Input
                placeholder="tag1, tag2, tag3"
                value={newSecretTags}
                onChange={(e) => setNewSecretTags(e.target.value)}
              />
              <p className="text-xs text-muted-foreground">Separate tags with commas</p>
            </div>
          </div>

          <DialogFooter>
            <Button variant="outline" onClick={() => setShowNewSecretDialog(false)}>
              Cancel
            </Button>
            <Button onClick={handleAddSecret}>
              <Plus className="w-4 h-4 mr-2" />
              Add Secret
            </Button>
          </DialogFooter>
        </DialogContent>
      </Dialog>

      {/* Import Dialog */}
      <Dialog open={showImportDialog} onOpenChange={setShowImportDialog}>
        <DialogContent className="max-w-2xl">
          <DialogHeader>
            <DialogTitle>Import Secrets</DialogTitle>
            <DialogDescription>
              Import secrets from JSON or .env format
            </DialogDescription>
          </DialogHeader>
          
          <div className="space-y-4">
            <div className="space-y-2">
              <Label>Format</Label>
              <Select value={importFormat} onValueChange={(val: "json" | "env") => setImportFormat(val)}>
                <SelectTrigger>
                  <SelectValue />
                </SelectTrigger>
                <SelectContent>
                  <SelectItem value="json">JSON</SelectItem>
                  <SelectItem value="env">.env</SelectItem>
                </SelectContent>
              </Select>
            </div>
            <div className="space-y-2">
              <Label>Data</Label>
              <Textarea
                placeholder={importFormat === "json" 
                  ? '{\n  "API_KEY": "your-key-here",\n  "DATABASE_URL": "postgres://..."\n}'
                  : 'API_KEY=your-key-here\nDATABASE_URL=postgres://...'
                }
                value={importData}
                onChange={(e) => setImportData(e.target.value)}
                className="font-mono text-sm h-64"
              />
            </div>
          </div>

          <DialogFooter>
            <Button variant="outline" onClick={() => setShowImportDialog(false)}>
              Cancel
            </Button>
            <Button onClick={handleImport}>
              <Upload className="w-4 h-4 mr-2" />
              Import
            </Button>
          </DialogFooter>
        </DialogContent>
      </Dialog>

      {/* Export Dialog */}
      <Dialog open={showExportDialog} onOpenChange={setShowExportDialog}>
        <DialogContent className="max-w-2xl">
          <DialogHeader>
            <DialogTitle>Export Secrets</DialogTitle>
            <DialogDescription>
              Export all secrets to JSON or .env format
            </DialogDescription>
          </DialogHeader>
          
          <div className="space-y-4">
            <div className="space-y-2">
              <Label>Format</Label>
              <Select value={exportFormat} onValueChange={(val: "json" | "env") => setExportFormat(val)}>
                <SelectTrigger>
                  <SelectValue />
                </SelectTrigger>
                <SelectContent>
                  <SelectItem value="json">JSON</SelectItem>
                  <SelectItem value="env">.env</SelectItem>
                </SelectContent>
              </Select>
            </div>
            <Alert>
              <Info className="w-4 h-4" />
              <AlertDescription>
                Exporting will copy {appSecrets.length} secrets to your clipboard in {exportFormat.toUpperCase()} format.
              </AlertDescription>
            </Alert>
          </div>

          <DialogFooter>
            <Button variant="outline" onClick={() => setShowExportDialog(false)}>
              Cancel
            </Button>
            <Button onClick={handleExport}>
              <Download className="w-4 h-4 mr-2" />
              Export to Clipboard
            </Button>
          </DialogFooter>
        </DialogContent>
      </Dialog>

      {/* Version History Dialog */}
      <Dialog open={showVersionDialog} onOpenChange={setShowVersionDialog}>
        <DialogContent className="max-w-3xl">
          <DialogHeader>
            <DialogTitle>Version History: {selectedSecret?.key}</DialogTitle>
            <DialogDescription>
              View and restore previous versions of this secret
            </DialogDescription>
          </DialogHeader>
          
          <ScrollArea className="h-96">
            <div className="space-y-3">
              {selectedSecret?.versions?.map((version, idx) => (
                <div 
                  key={idx}
                  className="flex items-start justify-between p-4 border rounded-lg"
                >
                  <div className="flex-1 space-y-2">
                    <div className="flex items-center gap-3">
                      <Badge variant={idx === 0 ? "default" : "secondary"}>
                        v{version.version}
                      </Badge>
                      {idx === 0 && (
                        <Badge variant="outline" className="text-green-600 border-green-600">
                          Current
                        </Badge>
                      )}
                    </div>
                    <div className="space-y-1">
                      <div className="flex items-center gap-2 text-sm">
                        <Calendar className="w-3 h-3 text-muted-foreground" />
                        <span className="text-muted-foreground">{version.timestamp}</span>
                      </div>
                      <div className="flex items-center gap-2 text-sm">
                        <span className="text-muted-foreground">Changed by:</span>
                        <span>{version.changedBy}</span>
                      </div>
                      <div className="font-mono text-sm bg-muted p-2 rounded mt-2">
                        {visibleSecrets.has(`${selectedSecret.id}-v${version.version}`) 
                          ? version.value 
                          : "•".repeat(version.value.length)}
                      </div>
                    </div>
                  </div>
                  <div className="flex gap-2 ml-4">
                    <Button
                      variant="ghost"
                      size="sm"
                      onClick={() => {
                        const key = `${selectedSecret.id}-v${version.version}`;
                        setVisibleSecrets(prev => {
                          const next = new Set(prev);
                          if (next.has(key)) {
                            next.delete(key);
                          } else {
                            next.add(key);
                          }
                          return next;
                        });
                      }}
                    >
                      <Eye className="w-4 h-4" />
                    </Button>
                    {idx !== 0 && (
                      <Button
                        variant="outline"
                        size="sm"
                        onClick={() => handleRestoreVersion(version)}
                      >
                        Restore
                      </Button>
                    )}
                  </div>
                </div>
              ))}
            </div>
          </ScrollArea>

          <DialogFooter>
            <Button variant="outline" onClick={() => setShowVersionDialog(false)}>
              Close
            </Button>
          </DialogFooter>
        </DialogContent>
      </Dialog>
    </div>
  );
}
