/**
 * Debugger Panel
 * Chrome DevTools Protocol integration for debugging
 * 
 * Features:
 * - Set/remove breakpoints
 * - Step through code
 * - View variables and call stack
 * - Console integration
 * - Watch expressions
 */

import { useState } from 'react';
import { Button } from './ui/button';
import { 
  Play, 
  Pause, 
  StepOver,
  StepInto,
  StepOut,
  Square,
  Bug,
  Eye,
  Plus,
} from 'lucide-react';
import { Card, CardContent, CardHeader, CardTitle } from './ui/card';
import { Input } from './ui/input';
import { ScrollArea } from './ui/scroll-area';
import { Tabs, TabsContent, TabsList, TabsTrigger } from './ui/tabs';
import { Badge } from './ui/badge';

interface Breakpoint {
  file: string;
  line: number;
  enabled: boolean;
}

interface Variable {
  name: string;
  value: string;
  type: string;
}

export function DebuggerPanel() {
  const [debugging, setDebugging] = useState(false);
  const [paused, setPaused] = useState(false);
  const [breakpoints, setBreakpoints] = useState<Breakpoint[]>([]);
  const [variables, setVariables] = useState<Variable[]>([]);
  const [watchExpressions, setWatchExpressions] = useState<string[]>([]);
  const [newWatch, setNewWatch] = useState('');

  const addWatch = () => {
    if (newWatch.trim()) {
      setWatchExpressions([...watchExpressions, newWatch]);
      setNewWatch('');
    }
  };

  return (
    <div className="h-full flex flex-col bg-background">
      {/* Header */}
      <div className="flex items-center justify-between px-4 py-2 bg-muted/30 border-b">
        <div className="flex items-center gap-2">
          <Bug className="w-4 h-4 text-muted-foreground" />
          <h3 className="font-medium">Debugger</h3>
          {debugging && (
            <Badge variant={paused ? 'destructive' : 'default'}>
              {paused ? 'Paused' : 'Running'}
            </Badge>
          )}
        </div>

        <div className="flex items-center gap-1">
          {!debugging ? (
            <Button size="sm" onClick={() => setDebugging(true)}>
              <Play className="w-4 h-4 mr-1" />
              Start Debugging
            </Button>
          ) : (
            <>
              {paused ? (
                <>
                  <Button size="sm" variant="ghost" onClick={() => setPaused(false)} title="Continue">
                    <Play className="w-4 h-4" />
                  </Button>
                  <Button size="sm" variant="ghost" title="Step Over">
                    <StepOver className="w-4 h-4" />
                  </Button>
                  <Button size="sm" variant="ghost" title="Step Into">
                    <StepInto className="w-4 h-4" />
                  </Button>
                  <Button size="sm" variant="ghost" title="Step Out">
                    <StepOut className="w-4 h-4" />
                  </Button>
                </>
              ) : (
                <Button size="sm" variant="ghost" onClick={() => setPaused(true)} title="Pause">
                  <Pause className="w-4 h-4" />
                </Button>
              )}
              <Button size="sm" variant="destructive" onClick={() => { setDebugging(false); setPaused(false); }}>
                <Square className="w-4 h-4 mr-1" />
                Stop
              </Button>
            </>
          )}
        </div>
      </div>

      {/* Content */}
      <div className="flex-1 overflow-hidden">
        <Tabs defaultValue="variables" className="h-full flex flex-col">
          <TabsList className="w-full justify-start border-b rounded-none bg-transparent px-4">
            <TabsTrigger value="variables">Variables</TabsTrigger>
            <TabsTrigger value="watch">Watch</TabsTrigger>
            <TabsTrigger value="breakpoints">Breakpoints</TabsTrigger>
            <TabsTrigger value="callstack">Call Stack</TabsTrigger>
          </TabsList>

          <TabsContent value="variables" className="flex-1 m-0">
            <ScrollArea className="h-full">
              <div className="p-4 space-y-2">
                {debugging && variables.length > 0 ? (
                  variables.map((v, i) => (
                    <Card key={i}>
                      <CardContent className="p-3">
                        <div className="flex justify-between items-start">
                          <div>
                            <code className="text-sm font-mono">{v.name}</code>
                            <p className="text-xs text-muted-foreground mt-1">{v.type}</p>
                          </div>
                          <code className="text-sm text-primary">{v.value}</code>
                        </div>
                      </CardContent>
                    </Card>
                  ))
                ) : (
                  <div className="text-center py-8 text-muted-foreground">
                    <Eye className="w-12 h-12 mx-auto mb-2 opacity-50" />
                    <p>No variables to display</p>
                    <p className="text-xs mt-1">Start debugging to view variables</p>
                  </div>
                )}
              </div>
            </ScrollArea>
          </TabsContent>

          <TabsContent value="watch" className="flex-1 m-0">
            <div className="p-4 space-y-4">
              <div className="flex gap-2">
                <Input
                  placeholder="Add watch expression..."
                  value={newWatch}
                  onChange={(e) => setNewWatch(e.target.value)}
                  onKeyDown={(e) => e.key === 'Enter' && addWatch()}
                />
                <Button onClick={addWatch} size="sm">
                  <Plus className="w-4 h-4" />
                </Button>
              </div>

              <ScrollArea className="h-[400px]">
                <div className="space-y-2">
                  {watchExpressions.map((expr, i) => (
                    <Card key={i}>
                      <CardContent className="p-3">
                        <code className="text-sm font-mono">{expr}</code>
                      </CardContent>
                    </Card>
                  ))}
                </div>
              </ScrollArea>
            </div>
          </TabsContent>

          <TabsContent value="breakpoints" className="flex-1 m-0">
            <ScrollArea className="h-full">
              <div className="p-4">
                {breakpoints.length === 0 ? (
                  <div className="text-center py-8 text-muted-foreground">
                    <p>No breakpoints set</p>
                    <p className="text-xs mt-1">Click line numbers in editor to add</p>
                  </div>
                ) : (
                  <div className="space-y-2">
                    {breakpoints.map((bp, i) => (
                      <Card key={i}>
                        <CardContent className="p-3">
                          <div className="flex items-center justify-between">
                            <div>
                              <code className="text-sm font-mono">{bp.file}</code>
                              <p className="text-xs text-muted-foreground">Line {bp.line}</p>
                            </div>
                            <Badge variant={bp.enabled ? 'default' : 'secondary'}>
                              {bp.enabled ? 'Enabled' : 'Disabled'}
                            </Badge>
                          </div>
                        </CardContent>
                      </Card>
                    ))}
                  </div>
                )}
              </div>
            </ScrollArea>
          </TabsContent>

          <TabsContent value="callstack" className="flex-1 m-0">
            <ScrollArea className="h-full">
              <div className="p-4">
                <div className="text-center py-8 text-muted-foreground">
                  <p>Call stack empty</p>
                  <p className="text-xs mt-1">Pause execution to view stack</p>
                </div>
              </div>
            </ScrollArea>
          </TabsContent>
        </Tabs>
      </div>
    </div>
  );
}
