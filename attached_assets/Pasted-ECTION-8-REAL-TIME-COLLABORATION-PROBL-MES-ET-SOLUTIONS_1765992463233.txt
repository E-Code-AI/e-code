8.1 RECONNEXION WEBSOCKET MANQUANTE
Fichier: client/src/components/collaboration/RealTimeCollaboration.tsx (lignes 143-154)

Solution:

const connectWithRetry = (attempts = 0) => {
  const provider = new WebsocketProvider(wsUrl, roomName, ydoc);
  
  provider.on('status', ({ status }) => {
    if (status === 'disconnected' && attempts < 10) {
      setTimeout(() => connectWithRetry(attempts + 1), 
        Math.min(1000 * Math.pow(2, attempts), 30000));
    }
  });
};

8.4 BUFFER OUTPUT TERMINAL MANQUANT
Fichier: server/terminal/pty-terminal-service.ts (lignes 170-175)

Solution:

class CircularBuffer {
  private buffer: string[] = [];
  private maxSize = 10000;
  
  push(data: string) {
    this.buffer.push(data);
    if (this.buffer.length > this.maxSize) {
      this.buffer.shift();
    }
  }
  
  getHistory(): string[] {
    return [...this.buffer];
  }
}

// Utiliser pour chaque session terminal
const outputBuffers = new Map<string, CircularBuffer>();

async resize(cols: number, rows: number) {
  await this.exec.kill();
  this.exec = await this.container.exec({
    Cmd: ['/bin/bash'],
    Tty: true,
    AttachStdin: true,
    AttachStdout: true,
    AttachStderr: true,
  });
  await this.exec.resize({ h: rows, w: cols });

8.7 SANDBOX ASYNC NON SUPPORTÉ
Fichier: server/execution/sandbox.ts (lignes 116-126)

Problème: Code exécuté de manière synchrone dans la VM.

Solution:

async executeAsync(code: string, timeout: number): Promise<ExecutionResult> {
  const wrappedCode = `
    (async () => {
      ${code}
    })()
  `;
  
  const script = new vm.Script(wrappedCode);
  const result = await script.runInContext(this.context, { timeout });
  return result;
}
