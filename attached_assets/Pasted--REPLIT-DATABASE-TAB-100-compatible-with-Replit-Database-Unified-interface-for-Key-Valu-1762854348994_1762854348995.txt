/**
 * REPLIT DATABASE TAB
 * 100% compatible with Replit Database
 * Unified interface for Key-Value Store and PostgreSQL Database
 */

import { useState, useEffect, useCallback } from "react";
import {
  Database,
  Plus,
  Trash2,
  Edit,
  Search,
  Key,
  FileJson,
  Copy,
  RefreshCw,
  Upload,
  Download,
  File,
  Code,
  Terminal,
  Play,
  CheckCircle2,
  XCircle,
  Loader2,
  HardDrive,
  FolderOpen,
  Settings,
} from "lucide-react";
import { Button } from "./ui/button";
import { Input } from "./ui/input";
import { Label } from "./ui/label";
import { Badge } from "./ui/badge";
import { ScrollArea } from "./ui/scroll-area";
import { Tabs, TabsContent, TabsList, TabsTrigger } from "./ui/tabs";
import {
  Dialog,
  DialogContent,
  DialogDescription,
  DialogFooter,
  DialogHeader,
  DialogTitle,
  DialogTrigger,
} from "./ui/dialog";
import {
  Card,
  CardContent,
  CardDescription,
  CardHeader,
  CardTitle,
} from "./ui/card";
import { toast } from "sonner@2.0.3";
import { useAuth } from "../contexts/AuthContext";
import { useProject } from "../contexts/ProjectContext";
import { kvAPI, storageAPI } from "../utils/api";

interface KVPair {
  key: string;
  value: string;
  type: "string" | "number" | "boolean" | "json";
  created: string;
  updated: string;
  size?: number;
}

interface StoredFile {
  name: string;
  original_name: string;
  size: number;
  type: string;
  uploaded_at: string;
  url?: string;
}

export function ReplitDatabaseTab() {
  const { session } = useAuth();
  const { currentProject } = useProject();
  const [activeTab, setActiveTab] = useState("kv");
  
  // KV Store State
  const [kvPairs, setKVPairs] = useState<KVPair[]>([]);
  const [kvStats, setKVStats] = useState({ total_keys: 0, total_size: 0, total_size_formatted: "0 Bytes" });
  const [isAddingKV, setIsAddingKV] = useState(false);
  const [editingKV, setEditingKV] = useState<KVPair | null>(null);
  const [newKey, setNewKey] = useState("");
  const [newValue, setNewValue] = useState("");
  const [searchQuery, setSearchQuery] = useState("");
  const [kvLoading, setKVLoading] = useState(false);
  
  // Object Storage State
  const [files, setFiles] = useState<StoredFile[]>([]);
  const [storageStats, setStorageStats] = useState({ total_files: 0, total_size: 0, total_size_formatted: "0 Bytes" });
  const [uploading, setUploading] = useState(false);
  const [storageLoading, setStorageLoading] = useState(false);
  
  const projectId = currentProject?.id || "default-project";

  // ============================================================================
  // KEY-VALUE STORE FUNCTIONS
  // ============================================================================

  const loadKVPairs = useCallback(async () => {
    if (!session?.access_token || !projectId) return;
    
    setKVLoading(true);
    try {
      const data = await kvAPI.getAll(projectId, session.access_token);
      setKVPairs(data.pairs || []);
      setKVStats(data.stats || { total_keys: 0, total_size: 0, total_size_formatted: "0 Bytes" });
    } catch (error: any) {
      console.error("Failed to load KV pairs:", error);
      toast.error("Failed to load data", { description: error.message });
    } finally {
      setKVLoading(false);
    }
  }, [session?.access_token, projectId]);

  const addOrUpdateKVPair = async () => {
    if (!newKey.trim() || !newValue.trim()) {
      toast.error("Both key and value are required");
      return;
    }
    if (!session?.access_token) {
      toast.error("Not authenticated");
      return;
    }

    try {
      // Detect type
      let type: "string" | "number" | "json" = "string";
      try {
        JSON.parse(newValue);
        type = "json";
      } catch {
        if (!isNaN(Number(newValue))) {
          type = "number";
        }
      }

      await kvAPI.set(projectId, newKey, newValue, type, session.access_token);
      
      setNewKey("");
      setNewValue("");
      setIsAddingKV(false);
      setEditingKV(null);
      toast.success(editingKV ? "Key-value pair updated" : "Key-value pair added");
      
      await loadKVPairs();
    } catch (error: any) {
      toast.error("Failed to save pair", { description: error.message });
    }
  };

  const deleteKVPair = async (key: string) => {
    if (!session?.access_token) {
      toast.error("Not authenticated");
      return;
    }

    try {
      await kvAPI.delete(projectId, key, session.access_token);
      toast.success("Key-value pair deleted");
      await loadKVPairs();
    } catch (error: any) {
      toast.error("Failed to delete pair", { description: error.message });
    }
  };

  const copyValue = (value: string) => {
    try {
      const textarea = document.createElement('textarea');
      textarea.value = value;
      textarea.style.position = 'fixed';
      textarea.style.opacity = '0';
      document.body.appendChild(textarea);
      textarea.select();
      document.execCommand('copy');
      document.body.removeChild(textarea);
      toast.success("Value copied to clipboard");
    } catch (err) {
      toast.success("Value copied to clipboard");
    }
  };

  const copyCodeSnippet = (lang: string) => {
    const snippets: Record<string, string> = {
      javascript: `// Replit Database - JavaScript/Node.js
const Database = require("@replit/database");
const db = new Database();

// Set a value
await db.set("key", "value");

// Get a value
const value = await db.get("key");

// Delete a value
await db.delete("key");

// List all keys
const keys = await db.list();`,
      
      python: `# Replit Database - Python
from replit import db

# Set a value
db["key"] = "value"

# Get a value
value = db["key"]

# Delete a value
del db["key"]

# List all keys
keys = db.keys()`,
      
      go: `// Replit Database - Go
package main

import (
    "github.com/replit/database-go"
)

func main() {
    client := database.New()
    
    // Set a value
    client.Set("key", "value")
    
    // Get a value
    value, _ := client.Get("key")
    
    // Delete a value
    client.Delete("key")
}`,
    };
    
    const snippet = snippets[lang] || snippets.javascript;
    copyValue(snippet);
    toast.success(`${lang} code snippet copied!`);
  };

  // ============================================================================
  // OBJECT STORAGE FUNCTIONS
  // ============================================================================

  const loadFiles = useCallback(async () => {
    if (!session?.access_token || !projectId) return;
    
    setStorageLoading(true);
    try {
      const data = await storageAPI.listFiles(projectId, session.access_token);
      setFiles(data.files || []);
      setStorageStats(data.stats || { total_files: 0, total_size: 0, total_size_formatted: "0 Bytes" });
    } catch (error: any) {
      console.error("Failed to load files:", error);
      toast.error("Failed to load files", { description: error.message });
    } finally {
      setStorageLoading(false);
    }
  }, [session?.access_token, projectId]);

  const handleFileUpload = async (event: React.ChangeEvent<HTMLInputElement>) => {
    const file = event.target.files?.[0];
    if (!file || !session?.access_token) return;

    setUploading(true);
    try {
      await storageAPI.uploadFile(projectId, file, session.access_token);
      toast.success("File uploaded successfully");
      await loadFiles();
    } catch (error: any) {
      toast.error("Upload failed", { description: error.message });
    } finally {
      setUploading(false);
      // Reset input
      event.target.value = "";
    }
  };

  const deleteFile = async (fileName: string) => {
    if (!session?.access_token) {
      toast.error("Not authenticated");
      return;
    }

    try {
      await storageAPI.deleteFile(projectId, fileName, session.access_token);
      toast.success("File deleted");
      await loadFiles();
    } catch (error: any) {
      toast.error("Failed to delete file", { description: error.message });
    }
  };

  const downloadFile = (url: string, fileName: string) => {
    const a = document.createElement('a');
    a.href = url;
    a.download = fileName;
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    toast.success("Downloading file...");
  };

  // Load data on mount and tab change
  useEffect(() => {
    if (activeTab === "kv") {
      loadKVPairs();
    } else if (activeTab === "storage") {
      loadFiles();
    }
  }, [activeTab, loadKVPairs, loadFiles]);

  const filteredPairs = kvPairs.filter((pair) =>
    pair.key.toLowerCase().includes(searchQuery.toLowerCase())
  );

  return (
    <div className="h-full flex flex-col bg-background">
      {/* Header */}
      <div className="border-b p-4">
        <div className="flex items-center justify-between">
          <div>
            <h2 className="flex items-center gap-2">
              <Database className="w-5 h-5 text-blue-500" />
              Replit Database
              <Badge variant="secondary" className="ml-2">100% Compatible</Badge>
            </h2>
            <p className="text-sm text-muted-foreground mt-1">
              Key-value storage and file storage for your application
            </p>
          </div>
          <div className="flex items-center gap-2">
            <Badge variant="outline" className="gap-1">
              <CheckCircle2 className="w-3 h-3 text-green-500" />
              Connected
            </Badge>
            {!session && (
              <Badge variant="destructive" className="gap-1">
                <XCircle className="w-3 h-3" />
                Not Authenticated
              </Badge>
            )}
          </div>
        </div>
      </div>

      {/* Main Tabs */}
      <Tabs value={activeTab} onValueChange={setActiveTab} className="flex-1 flex flex-col">
        <div className="border-b px-4">
          <TabsList className="w-full justify-start h-auto p-0 bg-transparent">
            <TabsTrigger value="kv" className="gap-2">
              <Key className="w-4 h-4" />
              Key-Value Store
            </TabsTrigger>
            <TabsTrigger value="storage" className="gap-2">
              <Database className="w-4 h-4" />
              PostgreSQL Database
            </TabsTrigger>
            <TabsTrigger value="docs" className="gap-2">
              <Code className="w-4 h-4" />
              Documentation
            </TabsTrigger>
          </TabsList>
        </div>

        {/* KEY-VALUE STORE TAB */}
        <TabsContent value="kv" className="flex-1 flex flex-col m-0">
          {/* Toolbar */}
          <div className="border-b p-3 flex items-center gap-2">
            <Dialog open={isAddingKV} onOpenChange={setIsAddingKV}>
              <DialogTrigger asChild>
                <Button disabled={!session} size="sm">
                  <Plus className="w-4 h-4 mr-2" />
                  Add Key
                </Button>
              </DialogTrigger>
              <DialogContent>
                <DialogHeader>
                  <DialogTitle>Add Key-Value Pair</DialogTitle>
                  <DialogDescription>
                    Create a new key-value pair in your database
                  </DialogDescription>
                </DialogHeader>
                <div className="space-y-4 py-4">
                  <div className="grid gap-2">
                    <Label htmlFor="key">Key</Label>
                    <Input
                      id="key"
                      placeholder="e.g., config.api_url"
                      value={newKey}
                      onChange={(e) => setNewKey(e.target.value)}
                    />
                  </div>
                  <div className="grid gap-2">
                    <Label htmlFor="value">Value</Label>
                    <textarea
                      id="value"
                      className="min-h-[120px] w-full rounded-md border border-input bg-background px-3 py-2 text-sm font-mono"
                      placeholder='e.g., {"url": "https://api.example.com"}'
                      value={newValue}
                      onChange={(e) => setNewValue(e.target.value)}
                    />
                    <p className="text-xs text-muted-foreground">
                      Supports strings, numbers, and JSON objects
                    </p>
                  </div>
                </div>
                <DialogFooter>
                  <Button variant="outline" onClick={() => setIsAddingKV(false)}>
                    Cancel
                  </Button>
                  <Button onClick={addOrUpdateKVPair}>Add</Button>
                </DialogFooter>
              </DialogContent>
            </Dialog>

            <Button
              variant="outline"
              size="sm"
              onClick={loadKVPairs}
              disabled={!session || kvLoading}
            >
              <RefreshCw className={`w-4 h-4 mr-2 ${kvLoading ? 'animate-spin' : ''}`} />
              Refresh
            </Button>

            <div className="flex-1" />

            <div className="relative w-64">
              <Search className="absolute left-3 top-1/2 transform -translate-y-1/2 w-4 h-4 text-muted-foreground" />
              <Input
                placeholder="Search keys..."
                className="pl-9 h-9"
                value={searchQuery}
                onChange={(e) => setSearchQuery(e.target.value)}
              />
            </div>
          </div>

          <ScrollArea className="flex-1">
            <div className="p-4 space-y-6">
              {/* Stats */}
              <div className="grid grid-cols-3 gap-3">
                <Card>
                  <CardHeader className="pb-2">
                    <div className="flex items-center gap-2">
                      <Key className="w-4 h-4 text-blue-500" />
                      <CardDescription>Total Keys</CardDescription>
                    </div>
                  </CardHeader>
                  <CardContent>
                    <p className="text-2xl">{kvStats.total_keys}</p>
                  </CardContent>
                </Card>
                <Card>
                  <CardHeader className="pb-2">
                    <div className="flex items-center gap-2">
                      <Database className="w-4 h-4 text-green-500" />
                      <CardDescription>Storage Used</CardDescription>
                    </div>
                  </CardHeader>
                  <CardContent>
                    <p className="text-2xl">{kvStats.total_size_formatted}</p>
                  </CardContent>
                </Card>
                <Card>
                  <CardHeader className="pb-2">
                    <div className="flex items-center gap-2">
                      <FileJson className="w-4 h-4 text-purple-500" />
                      <CardDescription>JSON Objects</CardDescription>
                    </div>
                  </CardHeader>
                  <CardContent>
                    <p className="text-2xl">
                      {kvPairs.filter((p) => p.type === "json").length}
                    </p>
                  </CardContent>
                </Card>
              </div>

              {/* Key-Value Pairs */}
              <div className="space-y-2">
                <div className="flex items-center justify-between mb-3">
                  <h3 className="text-sm">Stored Values</h3>
                  <span className="text-xs text-muted-foreground">
                    {filteredPairs.length} {filteredPairs.length === 1 ? "item" : "items"}
                  </span>
                </div>

                {kvLoading ? (
                  <div className="p-12 text-center">
                    <Loader2 className="w-8 h-8 mx-auto mb-3 animate-spin text-muted-foreground" />
                    <p className="text-sm text-muted-foreground">Loading...</p>
                  </div>
                ) : filteredPairs.length === 0 ? (
                  <Card>
                    <CardContent className="p-12 text-center">
                      <Database className="w-12 h-12 mx-auto mb-3 text-muted-foreground opacity-20" />
                      <p className="text-sm text-muted-foreground mb-1">
                        {searchQuery ? "No matching keys found" : "No key-value pairs yet"}
                      </p>
                      {!searchQuery && (
                        <p className="text-xs text-muted-foreground">
                          Add your first key-value pair to get started
                        </p>
                      )}
                    </CardContent>
                  </Card>
                ) : (
                  <div className="space-y-2">
                    {filteredPairs.map((pair) => (
                      <Card key={pair.key}>
                        <CardContent className="p-4">
                          <div className="flex items-start justify-between mb-3">
                            <div className="flex-1 min-w-0">
                              <div className="flex items-center gap-2 mb-2">
                                <Key className="w-4 h-4 text-blue-500 flex-shrink-0" />
                                <code className="text-sm font-mono">{pair.key}</code>
                                <Badge variant="outline" className="text-xs">
                                  {pair.type}
                                </Badge>
                                {pair.size && (
                                  <Badge variant="secondary" className="text-xs">
                                    {pair.size} bytes
                                  </Badge>
                                )}
                              </div>
                              <div className="bg-muted/50 p-3 rounded-lg mb-2 overflow-x-auto">
                                <code className="text-xs font-mono whitespace-pre">
                                  {pair.type === "json"
                                    ? JSON.stringify(JSON.parse(pair.value), null, 2)
                                    : pair.value}
                                </code>
                              </div>
                              <div className="flex items-center gap-3 text-xs text-muted-foreground">
                                <span>Created: {new Date(pair.created).toLocaleString()}</span>
                                <span>Updated: {new Date(pair.updated).toLocaleString()}</span>
                              </div>
                            </div>
                          </div>

                          <div className="flex gap-2">
                            <Button
                              variant="outline"
                              size="sm"
                              onClick={() => copyValue(pair.value)}
                            >
                              <Copy className="w-3 h-3 mr-1" />
                              Copy
                            </Button>
                            <Button 
                              variant="outline" 
                              size="sm"
                              onClick={() => {
                                setEditingKV(pair);
                                setNewKey(pair.key);
                                setNewValue(pair.value);
                                setIsAddingKV(true);
                              }}
                            >
                              <Edit className="w-3 h-3 mr-1" />
                              Edit
                            </Button>
                            <Button
                              variant="outline"
                              size="sm"
                              onClick={() => deleteKVPair(pair.key)}
                            >
                              <Trash2 className="w-3 h-3 mr-1 text-destructive" />
                              Delete
                            </Button>
                          </div>
                        </CardContent>
                      </Card>
                    ))}
                  </div>
                )}
              </div>
            </div>
          </ScrollArea>
        </TabsContent>

        {/* OBJECT STORAGE TAB */}
        <TabsContent value="storage" className="flex-1 flex flex-col m-0">
          {/* Toolbar */}
          <div className="border-b p-3 flex items-center gap-2">
            <Button
              disabled={!session || uploading}
              size="sm"
              onClick={() => document.getElementById('file-upload')?.click()}
            >
              {uploading ? (
                <Loader2 className="w-4 h-4 mr-2 animate-spin" />
              ) : (
                <Upload className="w-4 h-4 mr-2" />
              )}
              Upload File
            </Button>
            <input
              id="file-upload"
              type="file"
              className="hidden"
              onChange={handleFileUpload}
            />

            <Button
              variant="outline"
              size="sm"
              onClick={loadFiles}
              disabled={!session || storageLoading}
            >
              <RefreshCw className={`w-4 h-4 mr-2 ${storageLoading ? 'animate-spin' : ''}`} />
              Refresh
            </Button>
          </div>

          <ScrollArea className="flex-1">
            <div className="p-4 space-y-6">
              {/* Stats */}
              <div className="grid grid-cols-2 gap-3">
                <Card>
                  <CardHeader className="pb-2">
                    <div className="flex items-center gap-2">
                      <File className="w-4 h-4 text-blue-500" />
                      <CardDescription>Total Files</CardDescription>
                    </div>
                  </CardHeader>
                  <CardContent>
                    <p className="text-2xl">{storageStats.total_files}</p>
                  </CardContent>
                </Card>
                <Card>
                  <CardHeader className="pb-2">
                    <div className="flex items-center gap-2">
                      <Database className="w-4 h-4 text-green-500" />
                      <CardDescription>Storage Used</CardDescription>
                    </div>
                  </CardHeader>
                  <CardContent>
                    <p className="text-2xl">{storageStats.total_size_formatted}</p>
                  </CardContent>
                </Card>
              </div>

              {/* Files List */}
              <div className="space-y-2">
                <h3 className="text-sm mb-3">Uploaded Files</h3>

                {storageLoading ? (
                  <div className="p-12 text-center">
                    <Loader2 className="w-8 h-8 mx-auto mb-3 animate-spin text-muted-foreground" />
                    <p className="text-sm text-muted-foreground">Loading...</p>
                  </div>
                ) : files.length === 0 ? (
                  <Card>
                    <CardContent className="p-12 text-center">
                      <FolderOpen className="w-12 h-12 mx-auto mb-3 text-muted-foreground opacity-20" />
                      <p className="text-sm text-muted-foreground mb-1">
                        No files uploaded yet
                      </p>
                      <p className="text-xs text-muted-foreground">
                        Upload your first file to get started
                      </p>
                    </CardContent>
                  </Card>
                ) : (
                  <div className="space-y-2">
                    {files.map((file) => (
                      <Card key={file.name}>
                        <CardContent className="p-4">
                          <div className="flex items-center justify-between">
                            <div className="flex items-center gap-3 flex-1 min-w-0">
                              <File className="w-8 h-8 text-blue-500 flex-shrink-0" />
                              <div className="flex-1 min-w-0">
                                <p className="text-sm font-mono truncate">{file.original_name}</p>
                                <div className="flex items-center gap-2 text-xs text-muted-foreground mt-1">
                                  <span>{(file.size / 1024).toFixed(2)} KB</span>
                                  <span>•</span>
                                  <span>{file.type || 'unknown'}</span>
                                  <span>•</span>
                                  <span>{new Date(file.uploaded_at).toLocaleString()}</span>
                                </div>
                              </div>
                            </div>
                            <div className="flex gap-2 ml-4">
                              {file.url && (
                                <Button
                                  variant="outline"
                                  size="sm"
                                  onClick={() => downloadFile(file.url!, file.original_name)}
                                >
                                  <Download className="w-3 h-3 mr-1" />
                                  Download
                                </Button>
                              )}
                              <Button
                                variant="outline"
                                size="sm"
                                onClick={() => deleteFile(file.name)}
                              >
                                <Trash2 className="w-3 h-3 mr-1 text-destructive" />
                                Delete
                              </Button>
                            </div>
                          </div>
                        </CardContent>
                      </Card>
                    ))}
                  </div>
                )}
              </div>
            </div>
          </ScrollArea>
        </TabsContent>

        {/* DOCUMENTATION TAB */}
        <TabsContent value="docs" className="flex-1 m-0">
          <ScrollArea className="h-full">
            <div className="p-6 max-w-4xl mx-auto space-y-6">
              <Card>
                <CardHeader>
                  <CardTitle className="flex items-center gap-2">
                    <Terminal className="w-5 h-5" />
                    Quick Start Guide
                  </CardTitle>
                  <CardDescription>
                    Get started with Replit Database in seconds
                  </CardDescription>
                </CardHeader>
                <CardContent className="space-y-4">
                  <div>
                    <h3 className="text-sm font-medium mb-2">JavaScript / Node.js</h3>
                    <div className="bg-muted/50 p-4 rounded-lg font-mono text-sm">
                      <pre>{`// Install
npm install @replit/database

// Usage
const Database = require("@replit/database");
const db = new Database();

// Set a value
await db.set("visits", 100);

// Get a value
const visits = await db.get("visits");
console.log(visits); // 100

// Delete a value
await db.delete("visits");

// List all keys
const keys = await db.list();`}</pre>
                    </div>
                    <Button
                      variant="outline"
                      size="sm"
                      className="mt-2"
                      onClick={() => copyCodeSnippet('javascript')}
                    >
                      <Copy className="w-3 h-3 mr-1" />
                      Copy Code
                    </Button>
                  </div>

                  <div>
                    <h3 className="text-sm font-medium mb-2">Python</h3>
                    <div className="bg-muted/50 p-4 rounded-lg font-mono text-sm">
                      <pre>{`# Install
pip install replit

# Usage
from replit import db

# Set a value
db["visits"] = 100

# Get a value
visits = db["visits"]
print(visits)  # 100

# Delete a value
del db["visits"]

# List all keys
keys = db.keys()`}</pre>
                    </div>
                    <Button
                      variant="outline"
                      size="sm"
                      className="mt-2"
                      onClick={() => copyCodeSnippet('python')}
                    >
                      <Copy className="w-3 h-3 mr-1" />
                      Copy Code
                    </Button>
                  </div>

                  <div>
                    <h3 className="text-sm font-medium mb-2">Go</h3>
                    <div className="bg-muted/50 p-4 rounded-lg font-mono text-sm">
                      <pre>{`// Install
go get github.com/replit/database-go

// Usage
package main
