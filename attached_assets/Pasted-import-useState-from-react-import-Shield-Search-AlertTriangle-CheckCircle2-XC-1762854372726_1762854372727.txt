import { useState } from "react";
import {
  Shield,
  Search,
  AlertTriangle,
  CheckCircle2,
  XCircle,
  Info,
  Play,
  FileCode,
  Package,
  Lock,
} from "lucide-react";
import { Button } from "./ui/button";
import { Badge } from "./ui/badge";
import { ScrollArea } from "./ui/scroll-area";
import { Progress } from "./ui/progress";
import { Tabs, TabsContent, TabsList, TabsTrigger } from "./ui/tabs";
import { toast } from "sonner@2.0.3";

interface Vulnerability {
  id: string;
  severity: "critical" | "high" | "medium" | "low";
  title: string;
  description: string;
  file: string;
  line: number;
  recommendation: string;
}

export function SecurityScannerTab() {
  const [isScanning, setIsScanning] = useState(false);
  const [scanProgress, setScanProgress] = useState(0);

  const vulnerabilities: Vulnerability[] = [
    {
      id: "1",
      severity: "high",
      title: "SQL Injection Risk",
      description: "Unsanitized user input in database query",
      file: "src/api/users.ts",
      line: 45,
      recommendation: "Use parameterized queries or ORM",
    },
    {
      id: "2",
      severity: "medium",
      title: "Weak Password Requirements",
      description: "Password policy allows weak passwords",
      file: "src/auth/validation.ts",
      line: 23,
      recommendation: "Enforce minimum 8 characters with complexity",
    },
    {
      id: "3",
      severity: "low",
      title: "Missing CORS Configuration",
      description: "CORS not properly configured",
      file: "src/server.ts",
      line: 12,
      recommendation: "Configure CORS with whitelist",
    },
  ];

  const getSeverityColor = (severity: string) => {
    switch (severity) {
      case "critical":
        return "text-red-500 bg-red-500/10";
      case "high":
        return "text-orange-500 bg-orange-500/10";
      case "medium":
        return "text-yellow-500 bg-yellow-500/10";
      case "low":
        return "text-blue-500 bg-blue-500/10";
      default:
        return "text-gray-500 bg-gray-500/10";
    }
  };

  const getSeverityIcon = (severity: string) => {
    switch (severity) {
      case "critical":
      case "high":
        return <XCircle className="w-4 h-4" />;
      case "medium":
        return <AlertTriangle className="w-4 h-4" />;
      case "low":
        return <Info className="w-4 h-4" />;
      default:
        return <Info className="w-4 h-4" />;
    }
  };

  const startScan = () => {
    setIsScanning(true);
    setScanProgress(0);

    const interval = setInterval(() => {
      setScanProgress((prev) => {
        if (prev >= 100) {
          clearInterval(interval);
          setIsScanning(false);
          toast.success("Security scan completed");
          return 100;
        }
        return prev + 10;
      });
    }, 300);
  };

  const criticalCount = vulnerabilities.filter(v => v.severity === "critical").length;
  const highCount = vulnerabilities.filter(v => v.severity === "high").length;
  const mediumCount = vulnerabilities.filter(v => v.severity === "medium").length;
  const lowCount = vulnerabilities.filter(v => v.severity === "low").length;

  return (
    <div className="h-full flex flex-col bg-background">
      {/* Header */}
      <div className="border-b p-4">
        <h2 className="flex items-center gap-2">
          <Shield className="w-5 h-5" />
          Security Scanner
        </h2>
        <p className="text-sm text-muted-foreground mt-1">
          Scan your project for security vulnerabilities
        </p>
      </div>

      <ScrollArea className="flex-1">
        <div className="p-4 space-y-6">
          {/* Scan Controls */}
          <div className="p-4 border rounded-lg">
            <div className="flex items-center justify-between mb-4">
              <div>
                <h3 className="text-sm mb-1">Security Scan</h3>
                <p className="text-xs text-muted-foreground">
                  Last scan: 2 hours ago
                </p>
              </div>
              <Button onClick={startScan} disabled={isScanning}>
                {isScanning ? (
                  <>
                    <Search className="w-4 h-4 mr-2 animate-spin" />
                    Scanning...
                  </>
                ) : (
                  <>
                    <Play className="w-4 h-4 mr-2" />
                    Run Scan
                  </>
                )}
              </Button>
            </div>

            {isScanning && (
              <div className="space-y-2">
                <Progress value={scanProgress} />
                <p className="text-xs text-muted-foreground">
                  Scanning dependencies and code... {scanProgress}%
                </p>
              </div>
            )}
          </div>

          {/* Scan Results Summary */}
          <div className="grid grid-cols-4 gap-3">
            <div className="p-3 border rounded-lg">
              <div className="flex items-center gap-2 mb-2">
                <XCircle className="w-4 h-4 text-red-500" />
                <span className="text-xs text-muted-foreground">Critical</span>
              </div>
              <p className="text-2xl">{criticalCount}</p>
            </div>
            <div className="p-3 border rounded-lg">
              <div className="flex items-center gap-2 mb-2">
                <AlertTriangle className="w-4 h-4 text-orange-500" />
                <span className="text-xs text-muted-foreground">High</span>
              </div>
              <p className="text-2xl">{highCount}</p>
            </div>
            <div className="p-3 border rounded-lg">
              <div className="flex items-center gap-2 mb-2">
                <AlertTriangle className="w-4 h-4 text-yellow-500" />
                <span className="text-xs text-muted-foreground">Medium</span>
              </div>
              <p className="text-2xl">{mediumCount}</p>
            </div>
            <div className="p-3 border rounded-lg">
              <div className="flex items-center gap-2 mb-2">
                <Info className="w-4 h-4 text-blue-500" />
                <span className="text-xs text-muted-foreground">Low</span>
              </div>
              <p className="text-2xl">{lowCount}</p>
            </div>
          </div>

          {/* Vulnerabilities List */}
          <Tabs defaultValue="all">
            <TabsList className="grid w-full grid-cols-5">
              <TabsTrigger value="all">
                All ({vulnerabilities.length})
              </TabsTrigger>
              <TabsTrigger value="critical">Critical ({criticalCount})</TabsTrigger>
              <TabsTrigger value="high">High ({highCount})</TabsTrigger>
              <TabsTrigger value="medium">Medium ({mediumCount})</TabsTrigger>
              <TabsTrigger value="low">Low ({lowCount})</TabsTrigger>
            </TabsList>

            <TabsContent value="all" className="space-y-2 mt-4">
              {vulnerabilities.map((vuln) => (
                <div key={vuln.id} className="p-4 border rounded-lg">
                  <div className="flex items-start gap-3">
                    <div className="mt-0.5">
                      {getSeverityIcon(vuln.severity)}
                    </div>
                    <div className="flex-1">
                      <div className="flex items-center gap-2 mb-2">
                        <h4 className="text-sm">{vuln.title}</h4>
                        <Badge className={getSeverityColor(vuln.severity)}>
                          {vuln.severity.toUpperCase()}
                        </Badge>
                      </div>
                      <p className="text-sm text-muted-foreground mb-3">
                        {vuln.description}
                      </p>
                      <div className="flex items-center gap-2 text-xs text-muted-foreground mb-3">
                        <FileCode className="w-3 h-3" />
                        <code>{vuln.file}:{vuln.line}</code>
                      </div>
                      <div className="p-3 bg-muted rounded-lg">
                        <p className="text-xs mb-1">
                          <Lock className="w-3 h-3 inline mr-1" />
                          Recommendation:
                        </p>
                        <p className="text-xs text-muted-foreground">
                          {vuln.recommendation}
                        </p>
                      </div>
                    </div>
                  </div>
                  <div className="flex gap-2 mt-3">
                    <Button size="sm" variant="outline">
                      View Details
                    </Button>
                    <Button size="sm" variant="outline">
                      Fix Automatically
                    </Button>
                    <Button size="sm" variant="outline">
                      Ignore
                    </Button>
                  </div>
                </div>
              ))}
            </TabsContent>

            {["critical", "high", "medium", "low"].map((severity) => (
              <TabsContent key={severity} value={severity} className="space-y-2 mt-4">
                {vulnerabilities
                  .filter((v) => v.severity === severity)
                  .map((vuln) => (
                    <div key={vuln.id} className="p-4 border rounded-lg">
                      <div className="flex items-start gap-3">
                        <div className="mt-0.5">
                          {getSeverityIcon(vuln.severity)}
                        </div>
                        <div className="flex-1">
                          <div className="flex items-center gap-2 mb-2">
                            <h4 className="text-sm">{vuln.title}</h4>
                            <Badge className={getSeverityColor(vuln.severity)}>
                              {vuln.severity.toUpperCase()}
                            </Badge>
                          </div>
                          <p className="text-sm text-muted-foreground mb-3">
                            {vuln.description}
                          </p>
                          <div className="flex items-center gap-2 text-xs text-muted-foreground mb-3">
                            <FileCode className="w-3 h-3" />
                            <code>{vuln.file}:{vuln.line}</code>
                          </div>
                          <div className="p-3 bg-muted rounded-lg">
                            <p className="text-xs mb-1">
                              <Lock className="w-3 h-3 inline mr-1" />
                              Recommendation:
                            </p>
                            <p className="text-xs text-muted-foreground">
                              {vuln.recommendation}
                            </p>
                          </div>
                        </div>
                      </div>
                    </div>
                  ))}
                {vulnerabilities.filter((v) => v.severity === severity).length === 0 && (
                  <div className="p-8 text-center border rounded-lg">
                    <CheckCircle2 className="w-12 h-12 mx-auto mb-3 text-green-500 opacity-20" />
                    <p className="text-sm text-muted-foreground">
                      No {severity} severity issues found
                    </p>
                  </div>
                )}
              </TabsContent>
            ))}
          </Tabs>

          {/* Security Score */}
          <div className="p-4 border rounded-lg">
            <div className="flex items-center justify-between mb-3">
              <h3 className="text-sm">Security Score</h3>
              <Badge className="bg-green-500/10 text-green-500">
                B+ (Good)
              </Badge>
            </div>
            <Progress value={75} className="mb-2" />
            <p className="text-xs text-muted-foreground">
              75/100 - Address high severity issues to improve your score
            </p>
          </div>
        </div>
      </ScrollArea>
    </div>
  );
}
